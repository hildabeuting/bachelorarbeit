\begin{Definition}[Isometrie, unitärer Operator, vgl.\ {\cite[Def.~27.6, Def.~27.7]{serov}}] % S. 282
\label{def:isometrie}
Seien $\mathcal H_1$ und $\mathcal H_2$ Hilberträume. Eine lineare Abbildung
$U\colon \mathcal H_1 \to \mathcal H_2$ hei\ss{}t Isometrie, wenn
$\norm{Ux} = \norm{x}$ für alle $x \in \mathcal H_1$ gilt. Eine surjektive
Isometrie hei\ss{}t unitär.
\end{Definition}
 
Eine Isometrie erhält auch Skalarprodukte, also
$\scal{Ux}{Uy} = \scal{x}{y}$ für alle $x,y$ \cite[Übung~27.2]{serov}. % S. 282
Serov formuliert die Definition für $\mathcal H_1 = \mathcal H_2$. Wir
brauchen sie auch für verschiedene Räume, etwa für die Abbildung von $\LZwei$
in den Bildraum der Wavelet-Transformation (Satz~\ref{satz:plancherel-cwt}).

\begin{Bemerkung}[Fourier-Konvention]
\label{bem:fourier-konvention}
In dieser Arbeit wird durchgehend die unitäre Normierung
\[
  \hat f(\omega) \;=\; (2\pi)^{-1/2}\int_\R f(t)\,e^{-i\omega t}\,dt
\]
verwendet. Mit ihr ist die Fourier-Transformation ein unitärer Operator auf
$\LZwei$, und die Plancherel-Formel gilt ohne Vorfaktor:
$\scal{f}{g} = \scal{\hat f}{\hat g}$.

Die Literatur normiert uneinheitlich, und die Konstanten der folgenden Kapitel
hängen von der Wahl ab: Der Faktor $2\pi$ in der Zulässigkeitsbedingung
\eqref{eq:zulaessigkeit} und der Wert $\lvert\hat\varphi(0)\rvert = (2\pi)^{-1/2}$ in
Abschnitt~\ref{sec:skalierungsgleichung} gelten nur für diese Normierung. Beim Vergleich mit
einer Quelle ist deshalb zuerst deren Konvention nachzusehen.
\end{Bemerkung}

\label{sec:wahrscheinlichkeit}

In Kapitel~\ref{chapter:denoising} wird das Rauschen in einem Messsignal als
zufällig modelliert. Dafür genügen die folgenden Begriffe
\cite[Anhang~A.6]{mallat}. % S. 761--762
Ein Zufallsvektor $X = (X_0,\dots,X_{N-1})$ besteht aus $N$ Zufallsvariablen
$X_n$. Hat eine Zufallsvariable $X$ die Wahrscheinlichkeitsdichte $p$, so hei\ss en
\[
  \mathbb E[X] \;=\; \int_\R x\,p(x)\,dx
  \qquad\text{und}\qquad
  \operatorname{Var}(X) \;=\; \mathbb E\bigl[\lvert X - \mathbb E[X]\rvert^2\bigr]
\]
ihr Erwartungswert und ihre Varianz; die Kovarianz zweier Zufallsvariablen ist
\[
  \operatorname{Cov}(X_1,X_2) \;=\; \mathbb E\bigl[(X_1-\mathbb E[X_1])\,(X_2-\mathbb E[X_2])\bigr].
\]
Ist $\operatorname{Cov}(X_1,X_2) = 0$, so hei\ss en $X_1$ und $X_2$ unkorreliert.
Der Erwartungswert ist linear. Mit $\mathbb P$ bezeichnen wir
Wahrscheinlichkeiten.

\begin{Definition}[Normalverteilung, vgl.\ {\cite[Kap.~11.2.2]{mallat}}] % S. 554
\label{def:normalverteilung}
Sei $\phi(x) = (2\pi)^{-1/2}e^{-x^2/2}$ die Gau\ss'sche Dichte. Eine
Zufallsvariable $X$ hei\ss t normalverteilt mit Erwartungswert $\mu$ und Varianz
$\sigma^2>0$, kurz $X\sim\mathcal N(\mu,\sigma^2)$, wenn sie die Dichte
$p(x) = \sigma^{-1}\phi\bigl((x-\mu)/\sigma\bigr)$ hat, und standardnormalverteilt
im Fall $\mu = 0$, $\sigma = 1$.
\end{Definition}

Gemeinsam normalverteilt hei\ss en Zufallsvariablen, die zusammen einen
Gau\ss'schen Zufallsvektor bilden. Für sie gilt:

\begin{Satz}[vgl.\ {\cite[Kap.~11.1.1]{mallat}}] % S. 538, dort für zwei Zufallsvektoren
\label{satz:unkorreliert-unabhaengig}
Gemeinsam normalverteilte Zufallsvariablen, die paarweise unkorreliert sind,
sind unabhängig.
\end{Satz}

\subsection{Wavelets und die Zulässigkeitsbedingung}
\label{subsec:zulaessigkeit}
 
\anmerkung{davor auf jeden fall l2 norm definienren}
 
\begin{Definition}[Wavelet, {\cite[Kap.~3.1, Bedingungen (1) und (2)]{blatter}}] % S. 54
\label{def:wavelet}
Eine Funktion $\psi \in \LZwei$ mit $\norm{\psi}_{\LZwei} = 1$ hei\ss{}t
Mutter-Wavelet (kurz: Wavelet), falls sie die Zulässigkeitsbedingung
\begin{equation}
\label{eq:zulaessigkeit}
  C_\psi \;:=\; 2\pi \int_{\R^*} \frac{\lvert \hat\psi(\omega)\rvert^2}{\lvert\omega\rvert}\, d\omega \;<\; \infty
\end{equation}
erfüllt. Dabei ist $\R^* := \R\setminus\{0\}$.
\end{Definition}

Ob ein gegebenes $\psi$ die Bedingung \eqref{eq:zulaessigkeit} erfüllt, ist
nicht direkt zu sehen \cite[Kap.~3.1]{blatter}. % S. 54
Der folgende Satz führt sie auf eine einfache Bedingung im Zeitbereich zurück.

\begin{Satz}[Charakterisierung der Zulässigkeit, {\cite[Satz~(3.1)]{blatter}}] % S. 54
\label{satz:zulaessigkeit-mittelwert}
Sei $\psi \in \LZwei$, $\psi \neq 0$, mit $\int_\R \lvert t\rvert\,\lvert\psi(t)\rvert\,dt < \infty$.
Dann sind äquivalent:
\begin{enumerate}
\item[(i)] $C_\psi < \infty$,
\item[(ii)] $\displaystyle\int_\R \psi(t)\,dt = 0$.
\end{enumerate}
\end{Satz}

Der Beweis steht bei \cite[Satz~(3.1)]{blatter}. % S. 54--55
Ein Wavelet erfüllt also $\int_\R \psi(t)\,dt = 0$. Man sagt dafür auch, $\psi$
habe Mittelwert null \cite[Kap.~3.1]{blatter}. % S. 54
Sein Graph verläuft daher teils oberhalb, teils unterhalb der $t$-Achse.

\begin{Beispiel}[Haar-Wavelet]
\label{bsp:haar}
Die Funktion
\[
  \psi^{\mathrm{H}}(t) \;:=\;
  \begin{cases}
    \phantom{-}1, & 0 \le t < 1/2,\\
    -1, & 1/2 \le t < 1,\\
    \phantom{-}0, & \text{sonst,}
  \end{cases}
\]
hat kompakten Träger. Daher ist
$\int_\R \lvert t\rvert\,\lvert\psi^{\mathrm H}(t)\rvert\,dt < \infty$, und
Satz~\ref{satz:zulaessigkeit-mittelwert} ist anwendbar. Wegen
$\int_\R \psi^{\mathrm H}(t)\,dt = \tfrac12 - \tfrac12 = 0$ ist Bedingung (ii)
erfüllt. Also ist $\psi^{\mathrm H}$ zulässig. Mit
$\norm{\psi^{\mathrm H}}_{\LZwei} = 1$ ist $\psi^{\mathrm H}$ ein Wavelet im
Sinne von Definition~\ref{def:wavelet}.

Die Zulässigkeitskonstante lässt sich auch direkt berechnen. Mit
$\operatorname{sinc}(x) := \sin(x)/x$ für $x \neq 0$ und
$\operatorname{sinc}(0) := 1$ ist
\begin{equation}
\label{eq:haar-fourier}
  \widehat{\psi^{\mathrm{H}}}(\omega)
  \;=\; \frac{i}{\sqrt{2\pi}}\; e^{-i\omega/2}\,\sin(\omega/4)\,\operatorname{sinc}(\omega/4),
  \qquad \omega \in \R .
\end{equation}
Einsetzen in \eqref{eq:zulaessigkeit} ergibt $C_{\psi^{\mathrm{H}}} = 2\ln 2$
\cite[Kap.~1.1, Beispiel~(1)]{lmr} % S. 18--19
(Rechnung in Anhang~\ref{anh:haar-konstante}).
Abbildung~\ref{fig:haar-wavelet} zeigt $\psi^{\mathrm H}$ und
$\lvert\widehat{\psi^{\mathrm H}}\rvert$.
\end{Beispiel}

\subsection{Die Wavelet-Familie und die Transformation}
\label{subsec:wavelet-familie}
 
\begin{Definition}[Wavelet-Familie und kontinuierliche Wavelet-Transformation, {\cite[Kap.~3.1, Gl.~(4)--(6)]{blatter}}] % S. 55--56
\label{def:cwt}
Sei $\psi$ ein Wavelet im Sinne von Definition~\ref{def:wavelet}. Für
$(a,b) \in \R^* \times \R$ sei
\begin{equation}
\label{eq:wavelet-familie}
  \psi_{a,b}(t) \;:=\; \frac{1}{\lvert a\rvert^{1/2}}\,\psi\!\left(\frac{t-b}{a}\right),
  \qquad t \in \R .
\end{equation}
Die kontinuierliche Wavelet-Transformation (CWT) von $f \in \LZwei$ ist
\begin{equation}
\label{eq:cwt}
  W_\psi f \colon \R^*\times\R \to \C, \qquad
  W_\psi f(a,b) \;:=\; \scal{f}{\psi_{a,b}}
  \;=\; \frac{1}{\lvert a\rvert^{1/2}} \int_\R f(t)\,\overline{\psi\!\left(\frac{t-b}{a}\right)}\,dt .
\end{equation}
Die Menge $\R^*_\square := \R^*\times\R$ hei\ss{}t zersägte $(a,b)$-Ebene. Ist
$\psi$ aus dem Zusammenhang klar, schreiben wir kurz $Wf$.
\end{Definition}

%Der Wertebereich ist $\C$, weil Funktionen in $\LZwei$ komplexwertig sein
%dürfen. Daher steht in \eqref{eq:cwt} die komplexe Konjugation. Sind $f$ und
%$\psi$ reellwertig, so ist auch $W_\psi f$ reellwertig. Das gilt für alle
%Signale und Wavelets in dieser Arbeit.
 
\begin{Bemerkung}[Normierung der Familie]
\label{bem:normierung}
Es gilt $\norm{\psi_{a,b}}_{\LZwei} = \norm{\psi}_{\LZwei} = 1$ für alle
$(a,b)$. Einsetzen von \eqref{eq:wavelet-familie} und Quadrieren ergibt
\[
  \norm{\psi_{a,b}}_{\LZwei}^2
  \;=\; \int_\R \Bigl\lvert \lvert a\rvert^{-1/2}\,\psi\!\left(\tfrac{t-b}{a}\right)\Bigr\rvert^2 dt
  \;=\; \frac{1}{\lvert a\rvert}\int_\R \Bigl\lvert\psi\!\left(\tfrac{t-b}{a}\right)\Bigr\rvert^2 dt .
\]
Nun substituiert man $s = \frac{t-b}{a}$, also $t = as + b$ und $dt = a\,ds$.
Für $a < 0$ vertauscht die Substitution zusätzlich die Integrationsgrenzen. Das
gleicht das Vorzeichen aus, und es bleibt $dt = \lvert a\rvert\,ds$. Damit folgt
\[
  \frac{1}{\lvert a\rvert}\int_\R \Bigl\lvert\psi\!\left(\tfrac{t-b}{a}\right)\Bigr\rvert^2 dt
  \;=\; \frac{1}{\lvert a\rvert}\int_\R \lvert\psi(s)\rvert^2\,\lvert a\rvert\,ds
  \;=\; \int_\R \lvert\psi(s)\rvert^2\,ds
  \;=\; \norm{\psi}_{\LZwei}^2 .
\]
Mit einem Vorfaktor $\lvert a\rvert^{-p}$ statt $\lvert a\rvert^{-1/2}$ ergäbe
dieselbe Rechnung $\lvert a\rvert^{1-2p}\,\norm{\psi}_{\LZwei}^2$. Die Norm
hängt also genau für $p = \tfrac12$ nicht von $a$ ab \cite[Kap.~3.1]{blatter}. % S. 55
\end{Bemerkung}

\begin{Bemerkung}[Bedeutung der beiden Parameter]
\label{bem:parameter}
Ist $\psi$ kompakt getragen, so gilt $\supp \psi_{a,b} = b + a\cdot\supp\psi$.
Der Parameter $b$ verschiebt das Wavelet entlang der Zeitachse. Daher trägt
$Wf(a,b)$ Information über $f$ in der Nähe von $t = b$. Der Parameter $a$
streckt oder staucht das Wavelet und legt die Breite dieses Bereichs fest. Für
$a \to 0$ zieht sich $\psi_{a,b}$ auf $t = b$ zusammen \cite[Kap.~1.1]{lmr}. % S. 18
Abbildung~\ref{fig:haar-familie} zeigt beides am Haar-Wavelet.
\end{Bemerkung}

\paragraph{Wavelet-Transformierte.}
Wir verwenden das Haar-Wavelet und beschränken uns auf $a>0$. Dann ist
$\lvert a\rvert^{1/2} = \sqrt a$. Da $\psi^{\mathrm H}$ reell ist, entfällt die
Konjugation in \eqref{eq:cwt}, und es gilt
\[
  W_{\psi^{\mathrm H}}f(a,b)
  =\frac{1}{\sqrt a}\int_\R f(t)\,\psi^{\mathrm H}\Bigl(\frac{t-b}{a}\Bigr)\,dt .
\]
Das Fenster $[b,\,b+a[$ hat die Breite $a$. Auf seiner linken Hälfte
$[b,\,b+\tfrac a2[$ ist $\psi^{\mathrm H}\bigl(\frac{t-b}{a}\bigr) = 1$, auf
der rechten Hälfte $[b+\tfrac a2,\,b+a[$ ist es $-1$, sonst null. Also ist
\[
  W_{\psi^{\mathrm H}}f(a,b)
  =\frac{1}{\sqrt a}\left(\int_b^{b+a/2} f(t)\,dt-\int_{b+a/2}^{b+a} f(t)\,dt\right).
\]
Der Wert ist gro\ss{}, wenn sich die Integrale über die beiden Fensterhälften
stark unterscheiden. Ist $f$ auf dem Fenster nahezu konstant, ist er nahe null.
Mit der Stammfunktion $F(t):=\int_{-\infty}^{t}f(s)\,ds$
(Anhang~\ref{anh:stammfunktion-testsignal}) folgt die geschlossene Form
\begin{equation}\label{eq:cwt-haar-exakt}
  W_{\psi^{\mathrm H}}f(a,b)
  =\frac{2F\bigl(b+\tfrac a2\bigr)-F(b)-F(b+a)}{\sqrt a}.
\end{equation}
Abbildung~\ref{fig:cwt} zeigt ihren Betrag.

\begin{figure}[htbp]
  \centering
  \includegraphics[width=\textwidth]{abb3_cwt_haar.png}
  \caption{Betrag $\lvert W_{\psi^{\mathrm H}}f(a,b)\rvert$ des Testsignals nach
    \eqref{eq:cwt-haar-exakt}. Waagerecht ist die Verschiebung $b$ aufgetragen,
    senkrecht die Skala $a$ in logarithmischer Teilung. Die Farbe gibt den
    Betrag an.}
  \label{fig:cwt}
\end{figure}

Kleine $a$ gehören zu hohen Frequenzen, also schnellen Schwingungen, gro\ss{}e
$a$ zu tiefen Frequenzen, also langsamen Schwingungen \cite[Kap.~1.5]{blatter}. % S. 14
Genau das zeigt Abbildung~\ref{fig:cwt}. Die langsame Schwingung über $[0,3]$
liefert gro\ss{}e Werte bei gro\ss{}en $a$, die schnelle über $[4,5]$ bei
kleinen $a$. Die Wavelet-Transformation zeigt also gleichzeitig, wann und mit
welcher Frequenz $f$ schwingt.
\end{Beispiel}
 
\subsection{Die Plancherel-Formel}
\label{subsec:plancherel}

Dieser Abschnitt zeigt, dass bei der CWT keine Information verloren geht, sie
aber viel mehr Werte liefert als nötig. Das begründet den Übergang zum
dyadischen Gitter in Abschnitt~\ref{sec:mra-axiome}.

Für die Fourier-Transformation gilt $\norm{\hat f} = \norm{f}$
(Bemerkung~\ref{bem:fourier-konvention}). Für die CWT gilt eine ähnliche
Formel.

\begin{Satz}[Plancherel-Formel für die CWT, {\cite[Satz~(3.3)]{blatter}}] % S. 62
\label{satz:plancherel-cwt}
Sei $\psi$ ein Wavelet mit Zulässigkeitskonstante $C_\psi$ gemä\ss{}
\eqref{eq:zulaessigkeit}. Dann gilt für jedes $f \in \LZwei$
\begin{equation}
\label{eq:plancherel-cwt}
  \int_{\R^*_\square} \lvert W_\psi f(a,b)\rvert^2\,\frac{da\,db}{a^2}
  \;=\; C_\psi\,\norm{f}_{\LZwei}^2 .
\end{equation}
\end{Satz}

\anmerkung{hab ich das mit R kästchen schon definert?}

Der Beweis steht bei \cite[Satz~(3.3)]{blatter}. % S. 62--63
Links steht das Integral über $\lvert W_\psi f\rvert^2$ auf der
$(a,b)$-Ebene, gewichtet mit $a^{-2}$. Es ist bis auf den Faktor $C_\psi$
gleich $\norm{f}^2$. Das Gewicht $a^{-2}$ wird in
Abschnitt~\ref{sec:mra-axiome} erklärt.

Daraus folgt, dass $f$ durch $W_\psi f$ eindeutig bestimmt ist
\cite[Satz~(3.3)]{blatter}. % S. 62
Ist nämlich $W_\psi f = W_\psi g$, so ist $W_\psi(f-g) = 0$. Nach
\eqref{eq:plancherel-cwt} folgt $\norm{f-g} = 0$, also $f = g$.

\begin{Bemerkung}[Redundanz der CWT]
\label{bem:redundanz}
Da $f$ durch $W_\psi f$ eindeutig bestimmt ist, geht bei der CWT keine
Information verloren. Blatter gibt auch eine Formel an, mit der sich $f$ aus
$W_\psi f$ zurückgewinnen lässt \cite[Kap.~3.3]{blatter}. % S. 65--68
Die Werte $W_\psi f(a,b)$ sind aber nicht frei. Nicht jede Funktion auf
$\R^*_\square$ ist die Transformierte eines Signals
\cite[Kap.~3.4]{blatter}. % S. 69--70
Diese Eigenschaft hei\ss{}t Redundanz. Anschaulich hängt $f$ von einer
Variablen ab, $W_\psi f$ von zweien.

Im Endlichdimensionalen sieht man das direkt. Für ein Signal $x \in \R^N$
liefern $N$ Skalen und $N$ Verschiebungen $N^2$ Koeffizienten. Jeder ist ein
Skalarprodukt mit $x$. Zusammengefasst gilt daher $c = Ax$ mit einer Matrix
$A \in \R^{N^2 \times N}$. Der Vektor $c$ liegt also in einem Teilraum des
$\R^{N^2}$ der Dimension höchstens $N$.

Das Thresholding in Kapitel~\ref{chapter:denoising} arbeitet dagegen mit den
Koeffizienten bezüglich einer Orthonormalbasis \cite[Kap.~11.2]{mallat}. % S. 548--549
Gesucht ist daher eine abzählbare Auswahl von Parametern $(a,b)$, für die
$\{\psi_{a,b}\}$ eine Orthonormalbasis von $\LZwei$ ist.
\end{Bemerkung}

\section{Das dyadische Gitter und die Multiskalenanalyse}
\label{sec:mra-axiome}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Einen Anhaltspunkt für die Wahl der Punkte gibt die Plancherel-Formel
\eqref{eq:plancherel-cwt}. Dort geht jeder Punkt $(a,b)$ mit dem Gewicht
$a^{-2}$ ein. Ersetzt man das Integral durch eine Summe über Gitterpunkte,
sollte jeder Gitterpunkt eine Zelle mit gleichem Gewicht vertreten
\cite[Kap.~4.3]{blatter}. % S. 91--93

Das leistet das Gitter $a = 2^j$, $b = k2^j$ mit $j,k \in \Z$. Der Punkt
$(2^j, k2^j)$ vertritt die Zelle
\[
  Z_{j,k} := [2^j,\,2^{j+1}] \times [k2^j,\,(k+1)2^j] .
\]
In $a$ reicht sie über eine Verdopplung der Skala, in $b$ über die Länge
$2^j$. Für ihr Gewicht gilt
\begin{equation}
\label{eq:zellengewicht}
  \int_{k2^j}^{(k+1)2^j}\int_{2^j}^{2^{j+1}} \frac{da\,db}{a^2}
  \;=\; 2^j\Bigl(\frac{1}{2^j} - \frac{1}{2^{j+1}}\Bigr)
  \;=\; \frac12 .
\end{equation}
Das Gewicht hängt weder von $j$ noch von $k$ ab
(Abbildung~\ref{fig:dyadisches-gitter}). Ob die zugehörigen Funktionen
$\psi_{2^j,k2^j}$ eine Orthonormalbasis bilden, hängt aber von $\psi$ ab. Das
klärt die Multiskalenanalyse in Definition~\ref{def:msa}.

\begin{Definition}[Dyadische Wavelet-Familie, {\cite[Kap.~5.1]{blatter}}] % S. 106
\label{def:dyadisch}
Sei $\psi$ ein Wavelet. Auf dem dyadischen Gitter
$\{(2^{j},k2^{j}) : j,k\in\Z\}$ sei
\begin{equation}
\label{eq:dyadische-familie}
  \psi_{j,k}(t) \;:=\; \psi_{2^{j},\,k2^{j}}(t)
  \;=\; 2^{-j/2}\,\psi\!\left(\frac{t}{2^{j}} - k\right),
  \qquad j,k \in \Z .
\end{equation}
Kleineres $j$ gehört zur feineren Skala.\footnote{Diese Konvention folgt
Blatter und Louis/Maa\ss{}/Rieder. Hanke-Bourgeois zählt umgekehrt
\cite[§~56]{hankebourgeois}.} % S. 434
\end{Definition}

Im Beispiel liegen die Räume $V_j$ ineinander und gehen durch Mittelung
auseinander hervor. Die folgende Definition fasst die Eigenschaften zusammen,
auf die es dabei ankommt.

\begin{Definition}[Multiskalenanalyse, {\cite[Kap.~5.1, (a)--(c)]{blatter}}] % S. 106
\label{def:msa}
Eine Multiskalenanalyse (MSA) von $\LZwei$ ist eine Folge $(V_j)_{j\in\Z}$
abgeschlossener Teilräume von $\LZwei$ mit folgenden Eigenschaften:
\begin{enumerate}
\item[(M1)] (Schachtelung)
  $\;\cdots \subset V_{2} \subset V_{1} \subset V_{0} \subset V_{-1} \subset V_{-2} \subset \cdots \subset \LZwei$,
\item[(M2)] (Separation)
  $\;\displaystyle\bigcap_{j\in\Z} V_j = \{0\}$,
\item[(M3)] (Vollständigkeit)
  $\;\displaystyle\overline{\bigcup_{j\in\Z} V_j} = \LZwei$,
\item[(M4)] (Skalierung) Für alle $j \in \Z$ gilt: $f \in V_j$ genau dann, wenn
  die Funktion $t \mapsto f(2^{j}t)$ in $V_0$ liegt,
\item[(M5)] (Erzeugende Funktion) Es gibt ein
  $\varphi \in \LZwei \cap L^1(\R)$,\footnote{Die Forderung
  $\varphi \in L^1(\R)$ folgt Blatter. Sie sichert, dass $\hat\varphi$ stetig
  ist. Das wird für \eqref{eq:H0} gebraucht.} sodass die Funktionen
  $t \mapsto \varphi(t - k)$, $k \in \Z$, eine Orthonormalbasis von $V_0$ bilden.
\end{enumerate}
Die Funktion $\varphi$ hei\ss{}t Skalierungsfunktion der MSA.
\end{Definition}

\begin{Bemerkung}[Zu den Axiomen, {\cite[Kap.~5.1]{blatter}}] % S. 106--107
\label{bem:msa-axiome}
\begin{enumerate}
\item[(a)] (M1)--(M3): Je kleiner $j$, desto feinere Strukturen enthält $V_j$.
  Anschaulich geht $V_j$ für $j \to -\infty$ gegen ganz $\LZwei$ und für
  $j \to +\infty$ gegen $\{0\}$. Genau formuliert wird das durch (M3) und (M2)
  \cite[Bem.~2.2.2]{lmr}. % S. 112--113
\item[(b)] (M4): Die $V_j$ sind skalierte Kopien desselben Raumes $V_0$.
  Deshalb genügt eine einzige Skalierungsfunktion $\varphi$, um alle $V_j$ festzulegen.
\item[(c)] (M5): Das Axiom liefert eine Orthonormalbasis, allerdings nur für
  den Grundraum $V_0$ \cite[Kap.~5.1, (c)]{blatter}. % S. 106
  Zusammen mit (M4) überträgt sie sich auf alle $V_j$
  (Satz~\ref{satz:onb-vj}).\footnote{Louis, Maa\ss{} und Rieder verlangen in
  (M5) nur eine Riesz-Basis \cite[Def.~2.2.1]{lmr}. % S. 111
  Wir folgen Blatter und fordern eine Orthonormalbasis.}
\end{enumerate}
\end{Bemerkung}

\begin{Beispiel}[Die Haar-MSA, {\cite[Kap.~5.1, Beispiel~1]{blatter}}] % S. 107
\label{bsp:haar-msa}
Die Räume $V_j$ aus Beispiel~\ref{bsp:haar-mittelung} bilden eine MSA mit der
Skalierungsfunktion $\varphi := \mathbf 1_{[0,1[}$. Die Skalierungsfunktion ist
also eine einzelne Stufe und nicht das Haar-Wavelet. Die Axiome lassen sich
einzeln prüfen.
\begin{itemize}
\item (M1): Eine Treppe der Stufenbreite $2^{j}$ ist auch eine Treppe der
  Stufenbreite $2^{j-1}$.
\item (M2): Liegt $f$ in allen $V_j$, so ist $f$ auf $[0,2^j[$ für jedes $j$
  konstant, also auf $[0,\infty[$. In $\LZwei$ geht das nur für $f = 0$.
  Dasselbe gilt auf $]-\infty,0[$.
\item (M3): Treppenfunktionen mit Sprungstellen der Form $k2^j$ liegen dicht in
  $\LZwei$ \cite[Kap.~5.1]{blatter}. % S. 107
  Ein Beispiel: $\mathbf 1_{[0,1/3[}$ springt bei $\tfrac13$ und liegt daher in
  keinem $V_j$. Die Funktion $\mathbf 1_{[0,5/16[}$ springt bei
  $\tfrac5{16} = 5\cdot2^{-4}$ und liegt in $V_{-4}$. Der Fehler ist
  \[
    \norm{\mathbf 1_{[0,1/3[} - \mathbf 1_{[0,5/16[}}_{\LZwei}^2
    = \int_{5/16}^{1/3} 1\,dt = \tfrac13 - \tfrac5{16} = \tfrac1{48} .
  \]
  Mit feineren Stufen wird er beliebig klein.
\item (M4): $f$ ist genau dann auf allen Intervallen $[k2^j,(k+1)2^j[$
  konstant, wenn $t \mapsto f(2^j t)$ auf allen Intervallen $[k,k+1[$ konstant
  ist. Etwa ist $f = \mathbf 1_{[0,2[} \in V_1$. Für $g(t) := f(2t)$ gilt
  $g(t) = 1$ genau für $0 \le 2t < 2$, also für $0 \le t < 1$. Somit ist
  $g = \mathbf 1_{[0,1[} \in V_0$.
\item (M5): Die Verschiebungen $\varphi(t - k) = \mathbf 1_{[k,k+1[}(t)$ haben
  disjunkte Träger und Norm $1$. Sie sind also orthonormal. Etwa ist
  $f(t) = 4\varphi(t) + 2\varphi(t-1) + 5\varphi(t-2) + 7\varphi(t-3)$.
\end{itemize}
\end{Beispiel}

In Beispiel~\ref{bsp:haar-mittelung} entsteht jede gröbere Näherung
$P_{j+1} f$ durch Mittelung aus $P_j f$. Das entspricht der Schachtelung (M1).
Was bei der Mittelung verloren geht, sind Vielfache gestreckter und
verschobener Haar-Wavelets \cite[Kap.~2.2.1]{lmr}. % S. 110--111
Diese Unterschiede werden in Abschnitt~\ref{sec:detailraeume} als Detailräume
$W_j$ gefasst. Aus vier Werten werden wieder vier Koeffizienten. Die Zerlegung
ist also frei von der Redundanz aus Bemerkung~\ref{bem:redundanz}.

\section{Skalierungsfunktion und Orthonormalbasis der Räume \texorpdfstring{$V_j$}{Vj}}
\label{sec:skalierungsfunktion}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
 
In diesem Abschnitt wird die Basis aus (M5) auf alle Skalen übertragen
(Satz~\ref{satz:onb-vj}). Au\ss{}erdem wird gezeigt, dass die
Orthogonalprojektionen $P_j$ für $j \to -\infty$ gegen die Identität und für
$j \to +\infty$ gegen null streben (Proposition~\ref{prop:approximation}).
Normen ohne Index sind im Folgenden Normen in $\LZwei$.
 
%-------------------------------------------------------------------------------
\subsection{Die Orthonormalbasis auf allen Skalen}
\label{subsec:onb-alle-skalen}
%-------------------------------------------------------------------------------
 
Sei $D_a f := \lvert a\rvert^{-1/2} f(\,\cdot/a)$ der Dilatationsoperator,
$a \in \R^*$. Nach (M4) liegt $f$ genau dann in $V_j$, wenn
$f = h(\,\cdot/2^{j})$ mit einem $h \in V_0$ ist. Da $V_j$ ein linearer Raum ist,
spielt der Vorfaktor keine Rolle. Es folgt
\begin{equation}
\label{eq:vj-d2j}
  V_j \;=\; D_{2^{j}}(V_0) \qquad\text{für alle } j \in \Z
\end{equation}
\cite[Kap.~5.1, (b)]{blatter}. % S. 106
Der Operator $D_{2^j}$ ist unitär \cite[Kap.~1.2]{lmr}. % S. 27
 
\begin{Satz}[Orthonormalbasis von $V_j$, {\cite[Kap.~5.1]{blatter}}] % S. 107
\label{satz:onb-vj}
Sei $(V_j)_{j\in\Z}$ eine MSA mit Skalierungsfunktion $\varphi$. Mit
\begin{equation}
\label{eq:phi-jk}
  \varphi_{j,k}(t) \;:=\; 2^{-j/2}\,\varphi\!\left(\frac{t}{2^{j}} - k\right),
  \qquad j,k \in \Z ,
\end{equation}
ist $\{\varphi_{j,k} : k \in \Z\}$ für jedes feste $j \in \Z$ eine
Orthonormalbasis von $V_j$.
\end{Satz}
 
Ein unitärer Operator erhält Skalarprodukte. Er bildet daher eine
Orthonormalbasis eines abgeschlossenen Teilraums auf eine Orthonormalbasis des
Bildes ab \cite[Def.~27.7, Satz~25.18]{serov}. % S. 282, 257--258
Nach \eqref{eq:vj-d2j} bildet $D_{2^j}$ die Basis $\{\varphi(\,\cdot-k)\}_k$
von $V_0$ auf $\{\varphi_{j,k}\}_k$ ab. Das beweist den Satz.

In Abschnitt~\ref{sec:onb-invarianz} ist es die Orthogonalität der DWT
(Lemma~\ref{lem:dwt-orthogonal}), die das Rauschen invariant lässt.}
 
%-------------------------------------------------------------------------------
\subsection{Die Approximationsoperatoren}
\label{subsec:approximationsoperatoren}
%-------------------------------------------------------------------------------
 
\begin{Definition}[Approximationsoperator, {\cite[Bem.~2.2.2\,(a)]{lmr}}] % S. 113
\label{def:projektion-pj}
Für $f \in \LZwei$ und $j \in \Z$ hei\ss{}en
$a_{j,k} := \scal{f}{\varphi_{j,k}}$, $k \in \Z$, die
Approximationskoeffizienten von $f$ auf der Skala $j$. Die Abbildung
$P_j \colon \LZwei \to V_j$,
\begin{equation}
\label{eq:pj}
  P_j f \;:=\; \sum_{k\in\Z} a_{j,k}\,\varphi_{j,k},
\end{equation}
hei\ss{}t Approximationsoperator.
\end{Definition}
 
Wir halten drei Eigenschaften fest.
\begin{enumerate}
\item[(i)] Da $\{\varphi_{j,k}\}_k$ eine Orthonormalbasis von $V_j$ ist,
  konvergiert \eqref{eq:pj} in $\LZwei$, und es gilt
  $\norm{P_j f}^2 = \sum_{k} \lvert a_{j,k}\rvert^2$
  \cite[Satz~25.18]{serov}. % S. 257--258
\item[(ii)] Für alle $l \in \Z$ ist
  $\scal{f - P_j f}{\varphi_{j,l}} = a_{j,l} - a_{j,l} = 0$. Also steht
  $f - P_j f$ senkrecht auf $V_j$, und $P_j$ ist die Orthogonalprojektion auf
  $V_j$.
\item[(iii)] Aus (ii) und dem Satz des Pythagoras folgt
  $\norm{f - g} \ge \norm{f - P_j f}$ für alle $g \in V_j$. Also ist $P_j f$
  die beste Näherung von $f$ in $V_j$
  \cite[Satz~31.6]{hankebourgeois}. % S. 281
\end{enumerate}
 
Für die Haar-MSA ist $\varphi_{j,k} = 2^{-j/2}\,\mathbf 1_{I_{j,k}}$ mit
$I_{j,k} := [k2^{j},(k+1)2^{j}[$. Dann ist
\[
  a_{j,k}\,\varphi_{j,k}
  \;=\; \Bigl(\frac{1}{2^{j}}\int_{I_{j,k}} f(t)\,dt\Bigr)\,\mathbf 1_{I_{j,k}} .
\]
$P_j$ ersetzt $f$ also auf jedem $I_{j,k}$ durch seinen Mittelwert
\cite[Kap.~2.2]{lmr}. % S. 115
Das ist die Aussage aus Beispiel~\ref{bsp:haar-mittelung}.
 
\begin{Proposition}[Approximationseigenschaft, {\cite[Bem.~2.2.2\,(a)]{lmr}}] % S. 112--113
\label{prop:approximation}
Für jedes $f \in \LZwei$ gilt
\[
  \text{(a)}\;\; \lim_{j\to-\infty} \norm{P_j f - f} = 0
  \qquad\text{und}\qquad
  \text{(b)}\;\; \lim_{j\to+\infty} \norm{P_j f} = 0 .
\]
\end{Proposition}
 
Aussage (a) folgt aus (M3) und Eigenschaft (iii). Aussage (b) folgt aus (M2)
\cite[Prop.~(5.1)]{blatter}. % S. 108
 
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Detailräume \texorpdfstring{$W_j$}{Wj} und die Zerlegung von \texorpdfstring{$L^2(\R)$}{L2(R)}}
\label{sec:detailraeume}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
 
Wegen $V_j \subset V_{j-1}$ sind die Basen verschiedener $V_j$ nicht
zueinander orthogonal. Man betrachtet deshalb nur den Zuwachs beim Übergang
zur nächstfeineren Skala \cite[Kap.~5.1]{blatter}. % S. 107
 
\begin{Definition}[Detailräume, {\cite[Kap.~5.1, Gl.~(8)]{blatter}}] % S. 107
\label{def:detailraeume}
Für $j \in \Z$ sei $W_j$ das orthogonale Komplement von $V_j$ in $V_{j-1}$,
\begin{equation}
\label{eq:vw-zerlegung}
  V_{j-1} \;=\; V_j \oplus W_j,
  \qquad W_j \perp V_j ,
\end{equation}
und $Q_j := P_{j-1} - P_j$.
\end{Definition}
 
$Q_j$ ist die Orthogonalprojektion auf $W_j$ \cite[Kap.~5.1]{blatter}, % S. 109
\cite[Gl.~(2.2.9)]{lmr}. % S. 113
Denn $Q_j f = (f - P_j f) - (f - P_{j-1} f)$ liegt in $V_{j-1}$, und beide
Klammern stehen senkrecht auf $V_j$. Also liegt $Q_j f$ in $W_j$. Ferner steht
$f - Q_j f = (f - P_{j-1}f) + P_j f$ senkrecht auf $W_j$.
 
Da $D_{2^j}$ unitär ist und nach \eqref{eq:vj-d2j} $V_0$ auf $V_j$ und
$V_{-1}$ auf $V_{j-1}$ abbildet, gilt
\begin{equation}
\label{eq:wj-d2j}
  W_j \;=\; D_{2^{j}}(W_0) \qquad\text{für alle } j \in \Z
\end{equation}
\cite[Kap.~5.1, Gl.~(9)]{blatter}. % S. 107
Für $i > j$ ist $W_i \subset V_{i-1} \subset V_j \perp W_j$. Die Detailräume
sind also paarweise orthogonal \cite[Prop.~(5.1)]{blatter}. % S. 108
 
\begin{Satz}[Zerlegung von $\LZwei$ in Detailräume, {\cite[Prop.~(5.1)]{blatter}}] % S. 108
\label{satz:zerlegung}
Sei $(V_j)_{j\in\Z}$ eine MSA mit Detailräumen $W_j$. Dann gilt
\begin{equation}
\label{eq:l2-zerlegung}
  \LZwei \;=\; \bigoplus_{j\in\Z} W_j
\end{equation}
als orthogonale direkte Summe. Für jedes $f \in \LZwei$ ist
$f = \sum_{j\in\Z} Q_j f$ und
\begin{equation}
\label{eq:energie-skalen}
  \norm{f}^2 \;=\; \sum_{j\in\Z} \norm{Q_j f}^2 .
\end{equation}
Für jedes feste $J \in \Z$ gilt ferner
\begin{equation}
\label{eq:l2-zerlegung-endlich}
  \LZwei \;=\; V_J \oplus \bigoplus_{j \le J} W_j,
  \qquad
  f \;=\; P_J f + \sum_{j \le J} Q_j f .
\end{equation}
\end{Satz}
 
Nach Definition von $Q_j$ ist
$\sum_{j=J_1+1}^{J_2} Q_j f = P_{J_1} f - P_{J_2} f$. Nach
Proposition~\ref{prop:approximation} strebt das für $J_1 \to -\infty$ und
$J_2 \to +\infty$ gegen $f$. Mit $J_2 = J$ folgt
\eqref{eq:l2-zerlegung-endlich}. Die Energiegleichung folgt aus der
Orthogonalität der Summanden \cite[Prop.~(5.1)]{blatter}, % S. 108--109
\cite[Gl.~(2.2.11), (2.2.12)]{lmr}. % S. 114
 
In der Anwendung wird die endliche Fassung \eqref{eq:l2-zerlegung-endlich}
benutzt \cite[Kap.~2.2]{lmr}. % S. 114
Ein Mittelwert ungleich null verschwindet durch $P_j$ nämlich nicht, er wird nur
auseinandergezogen. Für die Treppenfunktion aus
Beispiel~\ref{bsp:haar-mittelung} ist etwa
$P_j f = 18\cdot 2^{-j}\,\mathbf 1_{[0,2^{j}[}$ für alle $j \ge 2$. In
Kapitel~\ref{chapter:anwendung} wird $P_J f$ dagegen gezielt entfernt, weil dort
die Basisliniendrift liegt (Abschnitt~\ref{sec:ein-schritt}).

\subsection{Die Wavelet-Basis}
\label{subsec:konstruktionsziel}
%-------------------------------------------------------------------------------
 
$V_{-1}$ besitzt zwei Basisfunktionen pro Zeiteinheit, $V_0$ eine. Es liegt
daher nahe, $W_0$ durch die ganzzahligen Verschiebungen einer einzigen
Funktion $\psi$ aufzuspannen \cite[Kap.~5.1]{blatter}. % S. 109
Dass es ein solches $\psi$ gibt, zeigt Kapitel~\ref{chapter:dwt}
\cite[Satz~2.2.10]{lmr}. % S. 122
Im Haar-Fall ist es $\psi^{\mathrm H}$ \cite[Kap.~2.2]{lmr}. % S. 116
 
\begin{Korollar}[Wavelet-Basis, {\cite[Kap.~5.1]{blatter}}] % S. 109--110
\label{kor:wavelet-basis}
Sei $(V_j)_{j\in\Z}$ eine MSA, und sei $\psi \in \LZwei$ so, dass
$\{\psi(\,\cdot-k) : k \in \Z\}$ eine Orthonormalbasis von $W_0$ ist. Mit
$\psi_{j,k}$ wie in \eqref{eq:dyadische-familie} gilt:
\begin{enumerate}
\item[(i)] Für jedes feste $j \in \Z$ ist $\{\psi_{j,k} : k \in \Z\}$ eine
  Orthonormalbasis von $W_j$.
\item[(ii)] Die Familie
  \begin{equation}
  \label{eq:wavelet-basis}
    \bigl\{\psi_{j,k} : j,k \in \Z\bigr\}
  \end{equation}
  ist eine Orthonormalbasis von $\LZwei$, die Wavelet-Basis der MSA.
\item[(iii)] Mit den Detailkoeffizienten $d_{j,k} := \scal{f}{\psi_{j,k}}$ gilt
  für jedes $f \in \LZwei$ und jedes $J \in \Z$
  \begin{equation}
  \label{eq:zerlegung-koeffizienten}
    f \;=\; \sum_{k\in\Z} a_{J,k}\,\varphi_{J,k}
         \;+\; \sum_{j\le J}\sum_{k\in\Z} d_{j,k}\,\psi_{j,k},
    \qquad
    \norm{f}^2 \;=\; \sum_{k\in\Z}\lvert a_{J,k}\rvert^2
                 \;+\; \sum_{j\le J}\sum_{k\in\Z}\lvert d_{j,k}\rvert^2 .
  \end{equation}
\end{enumerate}
\end{Korollar}
 
Aussage (i) folgt wie Satz~\ref{satz:onb-vj} aus \eqref{eq:wj-d2j}. Nach (i)
ist $Q_j f = \sum_k d_{j,k}\,\psi_{j,k}$. Damit folgen (ii) und (iii) aus
Satz~\ref{satz:zerlegung} und \eqref{eq:pj} \cite[Kap.~5.1]{blatter}, % S. 109--110
\cite[Gl.~(2.2.13)]{lmr}. % S. 114
 
\begin{Beispiel}[Wavelet-Koeffizienten der Treppenfunktion]
\label{bsp:haar-wavelet-koeff}
Wir zerlegen $f = (4,2,5,7)$ aus Beispiel~\ref{bsp:haar-mittelung} in der
Haar-Wavelet-Basis \cite[Kap.~2.2]{lmr}. % S. 116
Die Funktion $\psi_{j,k}$ ist auf der linken Hälfte von $I_{j,k}$ gleich
$2^{-j/2}$ und auf der rechten gleich $-2^{-j/2}$. Daher ist $d_{j,k}$ das
$2^{-j/2}$-Fache der Differenz der Integrale von $f$ über die beiden Hälften.
Für $J = 2$ erhält man
\begin{align*}
  a_{2,0} &= \tfrac12(4+2+5+7) = 9, &
  d_{2,0} &= \tfrac12\bigl((4+2)-(5+7)\bigr) = -3,\\
  d_{1,0} &= \tfrac{1}{\sqrt2}(4-2) = \sqrt2, &
  d_{1,1} &= \tfrac{1}{\sqrt2}(5-7) = -\sqrt2 .
\end{align*}
Alle übrigen Koeffizienten mit $j \le 2$ verschwinden. Nach
\eqref{eq:zerlegung-koeffizienten} ist
\[
  f \;=\; 9\,\varphi_{2,0} \;-\; 3\,\psi_{2,0} \;+\; \sqrt2\,\psi_{1,0} \;-\; \sqrt2\,\psi_{1,1},
  \qquad
  \norm{f}^2 \;=\; 81 + 9 + 2 + 2 \;=\; 94 .
\]
Bis auf den Faktor $2^{j/2}$ sind das die Zahlen $4{,}5$, $-1{,}5$, $1$ und $-1$
aus Beispiel~\ref{bsp:haar-mittelung}.
 
Zwei Beobachtungen verbinden das Beispiel mit den folgenden Kapiteln. Erstens
ist $d_{j,k} = W_\psi f(2^{j}, k2^{j})$. Die Detailkoeffizienten sind also
Werte der CWT auf dem dyadischen Gitter, aber ohne Redundanz. Zweitens bleibt
$9\,\varphi_{2,0} - 3\,\psi_{2,0} = P_1 f = (3,3,6,6)$, wenn man die beiden
betragskleinsten Koeffizienten $d_{1,0}$ und $d_{1,1}$ auf null setzt. Nach
der Energiegleichung ist der Fehler
$\norm{f - P_1 f}^2 = 2 + 2 = 4$, die Summe der Quadrate der gestrichenen
Koeffizienten. Welche Koeffizienten man streicht, lässt sich also im
Koeffizientenraum entscheiden. Darauf beruht die Entrauschung in
Kapitel~\ref{chapter:denoising}.
\end{Beispiel}
 
Gleichung~\eqref{eq:zerlegung-koeffizienten} ist der Ausgangspunkt der
folgenden Kapitel. Kapitel~\ref{chapter:dwt} konstruiert $\psi$ und berechnet
die Koeffizienten allein durch Filteroperationen. Der Schlüssel ist die
Inklusion $\varphi \in V_0 \subset V_{-1}$ \cite[Lemma~2.2.3]{lmr}, % S. 113
\cite[Satz~(5.3)]{blatter}. % S. 111
Kapitel~\ref{chapter:denoising} nutzt, dass $f \mapsto (a_{J,k}, d_{j,k})$
nach \eqref{eq:zerlegung-koeffizienten} eine Isometrie ist
(Definition~\ref{def:isometrie}).

\section{Die Skalierungsgleichung}
\label{sec:skalierungsgleichung}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\quellen{LMR, Lemma~2.2.3, S.~113 [bestätigt]; LMR, Gl.~(2.2.23), (2.2.24)
und Satz~2.2.9, S.~121 [bestätigt]; LMR, Anhang, S.~314 [bestätigt]; Serov,
Satz~25.18, S.~257--258 [bestätigt]; Blatter, Kap.~5.2, Gl.~(1), S.~110
[bestätigt], bewiesen in Satz~(5.8) [Seite nachtragen]; Blatter, Satz~(5.3),
S.~111 [bestätigt].}

In diesem Abschnitt wird die Skalierungsfunktion $\varphi$ durch eine
Zahlenfolge $(h_k)$ beschrieben, und es werden zwei Bedingungen an diese Folge
hergeleitet. Alles Weitere in diesem Kapitel wird nur noch mit den $h_k$
rechnen; die Funktionen $\varphi$ und $\psi$ selbst werden dafür nicht mehr
benötigt.

Nach (M1) ist $\varphi \in V_0 \subset V_{-1}$, und nach
Satz~\ref{satz:onb-vj} bilden die Funktionen
$\varphi_{-1,k}(t) = \sqrt2\,\varphi(2t-k)$ eine Orthonormalbasis von $V_{-1}$.
Also lässt sich $\varphi$ in dieser Basis darstellen:
\[
  \varphi = \sum_{k\in\Z} \langle \varphi, \varphi_{-1,k}\rangle\,
  \varphi_{-1,k}.
\]
Rechts steht eine gewichtete Summe der Funktionen $\varphi(2t-k)$. Jede davon
ist eine Kopie von $\varphi$, die um den Faktor $2$ zusammengedrückt und um
$k/2$ verschoben ist, denn $\varphi(2t-k) = \varphi\bigl(2(t-k/2)\bigr)$. Die
Skalierungsfunktion lässt sich also aus schmaleren Kopien ihrer selbst
zusammensetzen.

\begin{Satz}[Skalierungsgleichung, vgl.\ {\cite[Lemma~2.2.3]{lmr}}]
\label{satz:skalierungsgleichung}
Sei $(V_j)_{j\in\Z}$ eine MSA mit Skalierungsfunktion $\varphi$. Mit
$h_k := \scal{\varphi}{\varphi_{-1,k}}$ gilt
\begin{equation}
\label{eq:skalierungsgleichung}
  \varphi(t) \;=\; \sqrt2 \sum_{k\in\Z} h_k\,\varphi(2t-k)
  \quad\text{für fast alle } t\in\R
  \qquad\text{und}\qquad
  \sum_{k\in\Z}\lvert h_k\rvert^2 \;=\; 1 ,
\end{equation}
wobei die Reihe in $\LZwei$ konvergiert.
Die Folge $(h_k)$ hei\ss{}t Skalierungsfolge oder Tiefpassfilter der MSA.
\end{Satz}

\begin{proof}
Nach (M1) liegt $\varphi$ in $V_{-1}$, und $\{\varphi_{-1,k}\}_k$ ist nach
Satz~\ref{satz:onb-vj} eine Orthonormalbasis von $V_{-1}$. Die
Fourier-Entwicklung in dieser Basis \cite[Satz~25.18\,(3)]{serov} lautet
$\varphi = \sum_k h_k\,\varphi_{-1,k}$ mit
$h_k\,\varphi_{-1,k}(t) = \sqrt2\,h_k\,\varphi(2t-k)$. Nach der
Parseval-Gleichung \cite[Satz~25.18\,(5)]{serov} ist
$\sum_k\lvert h_k\rvert^2 = \norm{\varphi}^2 = 1$, denn $\varphi$ ist nach (M5)
selbst Element der Orthonormalbasis $\{\varphi(\,\cdot-k)\}_k$ von $V_0$.
\end{proof}

\begin{Beispiel}[Haar]
\label{bsp:haar-skalierung}
Für $\varphi = \mathbf 1_{[0,1[}$ aus Beispiel~\ref{bsp:haar-msa} ist
$\varphi(2t) = \mathbf 1_{[0,1/2[}(t)$ und
$\varphi(2t-1) = \mathbf 1_{[1/2,1[}(t)$. Eine Stufe der Breite $1$ besteht
also aus zwei Stufen der Breite $\tfrac12$:
\[
  \varphi(t) \;=\; \varphi(2t) + \varphi(2t-1)
  \;=\; \sqrt2\Bigl(\tfrac{1}{\sqrt2}\,\varphi(2t) + \tfrac{1}{\sqrt2}\,\varphi(2t-1)\Bigr).
\]
Der Vergleich mit \eqref{eq:skalierungsgleichung} liefert
$h_0 = h_1 = \tfrac{1}{\sqrt2}$ und $h_k = 0$ sonst, in Übereinstimmung mit
$h_0 = \scal{\varphi}{\varphi_{-1,0}} = \sqrt2\cdot\tfrac12$ und
$\sum_k\lvert h_k\rvert^2 = 1$.
\end{Beispiel}

Aus der Skalierungsgleichung folgt eine zweite Bedingung an $(h_k)$, die man im
Frequenzbereich sieht; dabei sei $\sum_k\lvert h_k\rvert<\infty$, was für
endliche Filter stets gilt. Nach den Rechenregeln der Fourier-Transformation für
Stauchung und Verschiebung \cite[Anhang]{lmr} ist
$\widehat{\varphi_{-1,k}}(\omega) = \tfrac{1}{\sqrt2}\,e^{-ik\omega/2}\,
\hat\varphi\bigl(\tfrac\omega2\bigr)$. Transformiert man
$\varphi = \sum_k h_k\,\varphi_{-1,k}$ gliedweise und zieht den von $k$
unabhängigen Faktor $\hat\varphi(\omega/2)$ vor die Summe, so wird die
Skalierungsgleichung mit dem $2\pi$-periodischen Fourier-Filter
\begin{equation}
\label{eq:symbol-H}
  H(\omega) \;:=\; \frac{1}{\sqrt2}\sum_{k\in\Z} h_k\,e^{-ik\omega}
\end{equation}
zu einem Produkt \cite[Gl.~(2.2.23)]{lmr}:
\begin{equation}
\label{eq:skalierung-fourier}
  \hat\varphi(\omega) \;=\; H\bigl(\tfrac\omega2\bigr)\,\hat\varphi\bigl(\tfrac\omega2\bigr),
  \qquad \omega\in\R .
\end{equation}
Für die hier betrachteten MSA ist $\lvert\hat\varphi(0)\rvert = (2\pi)^{-1/2}$,
insbesondere $\hat\varphi(0)\neq 0$ \cite[Kap.~5.2]{blatter}. Da beide Seiten
von \eqref{eq:skalierung-fourier} stetig sind, darf man $\omega = 0$ einsetzen;
Division durch $\hat\varphi(0)$ ergibt
\begin{equation}
\label{eq:H0}
  H(0) \;=\; 1, \qquad\text{das hei\ss{}t}\qquad \sum_{k\in\Z} h_k \;=\; \sqrt2 ,
\end{equation}
im Haar-Fall $\tfrac{1}{\sqrt2}+\tfrac{1}{\sqrt2} = \sqrt2$; vgl.\
\cite[Satz~2.2.9]{lmr}.

\section{Filterkoeffizienten: Tiefpass und Hochpass}
\label{sec:filter}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\quellen{LMR, Lemma~2.2.8, Gl.~(2.2.22) [Seite nachtragen, vermutl.\
S.~120--121]; LMR, Satz~2.2.10, S.~122, Beweis S.~122--123 [bestätigt]; LMR,
Beispiel S.~169 und Tabelle~2.3, S.~170 [bestätigt], dort umgekehrte
Reihenfolge wegen $e^{+ik\omega}$; Serov, Satz~25.18, S.~257--258 [bestätigt];
Blatter, Kap.~5.2, S.~110--116 [bestätigt].}

Bisher wurde die Folge $(h_k)$ aus einer gegebenen Skalierungsfunktion
gewonnen. Für die Konstruktion neuer Wavelets geht man umgekehrt vor: Man wählt
die Zahlen $h_k$ und erhält $\varphi$ und $\psi$ daraus. Dafür muss man wissen,
welche Bedingungen eine Folge erfüllen muss, um zu einer MSA zu gehören.
Neben \eqref{eq:H0} übersetzt sich auch die Orthonormalität der Verschiebungen
$\varphi(\,\cdot-k)$ aus (M5) in eine Bedingung an die Folge selbst.

\begin{Lemma}[Orthogonalität der Filterkoeffizienten, vgl.\ {\cite[Lemma~2.2.8]{lmr}}]
\label{lem:h-orthogonal}
Die Skalierungsfolge einer MSA erfüllt
\begin{equation}
\label{eq:h-orthogonal}
  \sum_{n\in\Z} h_n\,\overline{h_{n-2m}} \;=\; \delta_{m,0}
  \qquad\text{für alle } m\in\Z .
\end{equation}
\end{Lemma}

\begin{proof}
Ersetzt man in \eqref{eq:skalierungsgleichung} $t$ durch $t-m$ und setzt
$n = k+2m$, so wird
$\varphi(t-m) = \sqrt2\sum_k h_k\,\varphi(2t-2m-k) = \sum_n h_{n-2m}\,\varphi_{-1,n}(t)$.
Die Vollständigkeitsrelation in $V_{-1}$ \cite[Satz~25.18\,(4)]{serov} und (M5)
liefern
$\delta_{m,0} = \scal{\varphi}{\varphi(\,\cdot-m)} = \sum_n h_n\,\overline{h_{n-2m}}$.
\end{proof}

Anschaulich: Die Basisfunktionen $\varphi_{-1,n}$ liegen im Abstand
$\tfrac12$. Verschiebt man $\varphi$ um $1$, so rückt ihr Koeffizientenvektor
$(h_n)$ deshalb um zwei Stellen weiter. Die Orthogonalität der verschobenen
Funktionen wird so zur Orthogonalität des Vektors $(h_n)$ zu seinen um gerade
Schrittweiten verschobenen Kopien. Für das Haar-Filter ist das trivial, denn
für $m \neq 0$ überlappen $(h_n)$ und $(h_{n-2m})$ nicht.

\begin{Beispiel}[Ein Filter der Länge vier]
\label{bsp:db2-filter}
Gesucht seien reelle $h_0,\dots,h_3$, alle übrigen null. Die Bedingungen
\eqref{eq:H0} und \eqref{eq:h-orthogonal} lauten
\[
  h_0+h_1+h_2+h_3 = \sqrt2, \qquad
  h_0^2+h_1^2+h_2^2+h_3^2 = 1, \qquad
  h_0h_2+h_1h_3 = 0 ;
\]
für $\lvert m\rvert\ge2$ überlappen die Folgen nicht. Drei Gleichungen für vier
Unbekannte lassen eine einparametrige Lösungsschar, zu der auch das Haar-Filter
$(\tfrac{1}{\sqrt2},\tfrac1{\sqrt2},0,0)$ gehört. Eine weitere Lösung ist
\begin{equation}
\label{eq:db2-koeff}
  h_0 = \tfrac{1+\sqrt3}{4\sqrt2},\qquad
  h_1 = \tfrac{3+\sqrt3}{4\sqrt2},\qquad
  h_2 = \tfrac{3-\sqrt3}{4\sqrt2},\qquad
  h_3 = \tfrac{1-\sqrt3}{4\sqrt2}
\end{equation}
\cite[Tabelle~2.3]{lmr}, dort in umgekehrter Reihenfolge. Zur Probe: Die Zähler summieren sich zu $8$, also
$\sum h_k = 8/(4\sqrt2) = \sqrt2$. Die quadrierten Zähler ergeben
$(4+2\sqrt3)+(12+6\sqrt3)+(12-6\sqrt3)+(4-2\sqrt3) = 32 = (4\sqrt2)^2$, also
$\sum h_k^2 = 1$. Schlie\ss{}lich ist
$(1+\sqrt3)(3-\sqrt3)+(3+\sqrt3)(1-\sqrt3) = 2\sqrt3-2\sqrt3 = 0$. Warum gerade
diese Lösung aus der Schar ausgewählt wird, klärt
Abschnitt~\ref{sec:daubechies}.
\end{Beispiel}

Im Haar-Fall haben die Zahlen $h_k$ eine einfache Bedeutung: Mit
$h_0 = h_1 = \tfrac{1}{\sqrt2}$ bildet $(h_k)$ aus zwei benachbarten Werten
deren (normierten) Mittelwert, wie in Beispiel~\ref{bsp:haar-mittelung}.
Konstante Signale bleiben dabei erhalten, rasche Schwankungen werden
geglättet; daher der Name Tiefpass. Das Gegenstück, das in
Beispiel~\ref{bsp:haar-mittelung} die verlorenen Details liefert, ist die
Differenz benachbarter Werte. Sie verschwindet für konstante Signale und
hei\ss{}t Hochpass. Für allgemeine $(h_k)$ gewinnt man die Hochpassfolge
durch Umkehren der Reihenfolge und alternierende Vorzeichen.

\begin{Satz}[Wavelet aus der Skalierungsfolge, vgl.\ {\cite[Satz~2.2.10]{lmr}}]
\label{satz:wavelet-aus-h}
Sei $(V_j)$ eine MSA mit Skalierungsfolge $(h_k)$. Setzt man
\begin{equation}
\label{eq:g-aus-h}
  g_k \;:=\; (-1)^k\,\overline{h_{1-k}}, \qquad
  \psi(t) \;:=\; \sqrt2\sum_{k\in\Z} g_k\,\varphi(2t-k),
\end{equation}
so ist $\{\psi(\,\cdot-k) : k\in\Z\}$ eine Orthonormalbasis von $W_0$. Nach
Korollar~\ref{kor:wavelet-basis} ist dann $\{\psi_{j,k} : j,k\in\Z\}$ eine
Orthonormalbasis von $\LZwei$.
\end{Satz}

Der vollständige Beweis steht bei \cite[Satz~2.2.10]{lmr}.

\begin{Beispiel}[Haar, Fortsetzung]
\label{bsp:haar-g}
Aus $h_0=h_1=1/\sqrt2$ folgt $g_0 = h_1 = 1/\sqrt2$ und $g_1 = -h_0 = -1/\sqrt2$,
also
\[
  \psi(t) \;=\; \varphi(2t) - \varphi(2t-1) \;=\; \mathbf 1_{[0,1/2[}(t) - \mathbf 1_{[1/2,1[}(t)
  \;=\; \psi^{\mathrm H}(t).
\]
Damit ist $\psi^{\mathrm H}$ als Wavelet der Haar-MSA nachgewiesen, wie es
Beispiel~\ref{bsp:haar-mittelung} nahegelegt hatte. Für die Folge
\eqref{eq:db2-koeff} ist entsprechend
$(g_{-2},g_{-1},g_0,g_1) = (h_3,-h_2,h_1,-h_0)$.
\end{Beispiel}

\section{Bedingungen im Fourier-Bereich}
\label{sec:fourier-bedingungen}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\quellen{LMR, Satz~2.2.9, S.~121 [bestätigt]; LMR, Korollar~2.2.12, S.~123
[bestätigt]; LMR, Bem.~2.4.19, S.~165 [bestätigt]; LMR, Kap.~2.4.2,
S.~145--164 [bestätigt].}

In Abschnitt~\ref{sec:filter} wurden die Namen Tiefpass und Hochpass nur am
Haar-Filter begründet. Für beliebige Filter liefert das Fourier-Filter $H$ aus
\eqref{eq:symbol-H} die Begründung. Es gibt an, wie stark der Filter eine
Schwingung der Frequenz $\omega$ durchlässt: Für $x_n = e^{i\omega n}$ ist
$\sum_k h_k\,x_{n-k} = \sqrt2\,H(\omega)\,x_n$. Für $\omega = 0$ ist $x_n = 1$ konstant, für $\omega = \pi$ ist
$x_n = (-1)^n$, die Folge wechselt also bei jedem Schritt das Vorzeichen.
Schneller kann eine abgetastete Schwingung nicht oszillieren, da
$e^{i(\omega+2\pi)n} = e^{i\omega n}$ gilt. Wie in Abschnitt~\ref{sec:skalierungsgleichung} sei
$\sum_k\lvert h_k\rvert<\infty$.

\begin{Satz}[Bedingungen an das Fourier-Filter, {\cite[Satz~2.2.9]{lmr}}]
\label{satz:qmf}
Das Fourier-Filter \eqref{eq:symbol-H} einer MSA erfüllt
\begin{equation}
\label{eq:qmf}
  H(0) = 1 \qquad\text{und}\qquad
  \lvert H(\omega)\rvert^2 + \lvert H(\omega+\pi)\rvert^2 = 1
  \quad\text{für alle } \omega\in\R .
\end{equation}
\end{Satz}

Die erste Gleichung ist \eqref{eq:H0}. Die zweite ist
Lemma~\ref{lem:h-orthogonal}, in den Frequenzbereich übersetzt: Die unendlich
vielen Gleichungen für die Zahlen $h_k$ werden zu einer einzigen Gleichung
für die Funktion $H$. Umgekehrt liefert jede endliche Folge mit \eqref{eq:qmf},
deren Fourier-Filter auf $[-\pi/2,\pi/2]$ keine Nullstelle hat, eine Lösung
$\varphi$ der Skalierungsgleichung mit orthonormalen Verschiebungen
$\varphi(\,\cdot-k)$ \cite[Satz~2.4.7]{lmr}.

Für die Hochpassfolge $(g_k)$ aus \eqref{eq:g-aus-h} setzt man analog
$G(\omega) := 2^{-1/2}\sum_k g_k e^{-ik\omega}$. Die Substitution $n = 1-k$
ergibt
\begin{equation}
\label{eq:G-aus-H}
  G(\omega) \;=\; -e^{-i\omega}\,\overline{H(\omega+\pi)},
  \qquad
  \hat\psi(\omega) \;=\; G\bigl(\tfrac\omega2\bigr)\,\hat\varphi\bigl(\tfrac\omega2\bigr)
\end{equation}
\cite[Korollar~2.2.12]{lmr}, also $\lvert G(\omega)\rvert = \lvert H(\omega+\pi)\rvert$.
Aus \eqref{eq:qmf} und \eqref{eq:G-aus-H} folgen drei Aussagen:
\begin{enumerate}
\item[(a)] $H(0) = 1$ und $H(\pi) = 0$: $(h_k)$ lässt konstante Signale
  vollständig durch und löscht die schnellste Schwingung aus. Das ist ein Tiefpass.
\item[(b)] $G(0) = 0$ und $\lvert G(\pi)\rvert = 1$: $(g_k)$ verhält sich
  genau umgekehrt. Das ist ein Hochpass. Wegen $\hat\psi(0) = G(0)\,\hat\varphi(0) = 0$
  hat $\psi$ Mittelwert null, wie es Satz~\ref{satz:zulaessigkeit-mittelwert}
  von einem zulässigen Wavelet verlangt; zulässig ist $\psi$ tatsächlich stets
  \cite[Satz~2.2.10\,(iii)]{lmr}.
\item[(c)] $\lvert H(\omega)\rvert^2 + \lvert G(\omega)\rvert^2 = 1$: Jede
  Frequenz wird vollständig auf die beiden Filter aufgeteilt, es geht nichts
  verloren. In der Signalverarbeitung hei\ss{}en solche Filterpaare
  Conjugate Quadrature Filter \cite[Bem.~2.4.19]{lmr}.
\end{enumerate}

Abbildung~\ref{fig:tief-hochpass} zeigt beide Filter. Beim Haar-Filter ist
$\lvert H(\omega)\rvert = \lvert\cos(\omega/2)\rvert$ und die Nullstelle bei
$\omega=\pi$ einfach. Für \eqref{eq:db2-koeff} ist
$\lvert H(\omega)\rvert^2 = \cos^4(\omega/2)\,\bigl(1+2\sin^2(\omega/2)\bigr)$,
vgl.\ \cite[Kap.~2.4.3]{lmr}, die Nullstelle also doppelt, und die Kurven
verlaufen an den Rändern flacher. Abschnitt~\ref{sec:daubechies} zeigt, dass
genau diese Ordnung der Nullstelle bestimmt, wie gut sich ein Wavelet für
glatte Signalabschnitte eignet.

\section{Der Mallat-Algorithmus}
\label{sec:mallat}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\quellen{Blatter, Kap.~5.4, S.~130--136 [bestätigt], insb.\ Skalierungs-
gleichungen 5.4.(1)/(2), S.~130 [bestätigt], Zwei-Skalen-Relation
5.4.(3)/(4) mit Herleitung, S.~131 [bestätigt], Analyseformeln
(eingerahmt, unnummeriert), S.~132--133 [bestätigt]; LMR, Kap.~2.3,
S.~132--141 [bestätigt], insb.\ Analyseformeln (unnummeriert, mit Verweis
auf Skalierungsgl.~(2.2.8)/(2.2.27)), S.~133 [bestätigt],
Gl.~(2.3.1)/(2.3.2), S.~133 [bestätigt], Abb.~2.8, S.~134 [bestätigt];
Aufwand: Blatter, Kap.~5.4, S.~133--135 [bestätigt], Hanke-Bourgeois,
§~56, S.~438 [bestätigt]; LMR, Kap.~3.1.1, S.~237--238 [bestätigt].}

Bisher erfordert jeder Koeffizient
\[
  a_{j,k} = \scal{f}{\varphi_{j,k}} = \int_{\R} f(t)\,\varphi_{j,k}(t)\,dt,
  \qquad
  d_{j,k} = \scal{f}{\psi_{j,k}} = \int_{\R} f(t)\,\psi_{j,k}(t)\,dt
\]
(Definition~\ref{def:projektion-pj}, Korollar~\ref{kor:wavelet-basis}) ein
eigenes Integral. Nach \cite[Kap.~5.4]{blatter} und
\cite[Kap.~2.3, Abb.~2.8]{lmr} lassen sich $a_{j,\cdot}$ und $d_{j,\cdot}$
jedoch allein aus $a_{j-1,\cdot}$ und den Filtern $h$, $g$ berechnen.
Grundlage dafür ist die folgende Darstellung der Basisfunktionen der Skala
$j$ durch die der feineren Skala $j-1$.

\begin{Lemma}[Zwei-Skalen-Relation, {\cite[Kap.~5.4, (3), (4)]{blatter}}]
\label{lem:zwei-skalen}
Für alle $j,k\in\Z$ gilt
\begin{equation}
\label{eq:zwei-skalen}
  \varphi_{j,k} \;=\; \sum_{n\in\Z} h_{n-2k}\,\varphi_{j-1,n},
  \qquad
  \psi_{j,k} \;=\; \sum_{n\in\Z} g_{n-2k}\,\varphi_{j-1,n} .
\end{equation}
\end{Lemma}

Für einen Beweis siehe \cite[Kap.~5.4]{blatter}. Bildet man in
\eqref{eq:zwei-skalen} das Skalarprodukt mit $f$, so überträgt sich die
Relation auf die Koeffizienten.

\begin{Satz}[Analyseschritt, {\cite[Kap.~5.4]{blatter}}]
\label{satz:analyse}
Für reelle Filter $h,g$ und alle $j,k\in\Z$ gilt
\begin{equation}
\label{eq:analyse}
  a_{j,k} \;=\; \sum_{n\in\Z} h_{n-2k}\,a_{j-1,n},
  \qquad
  d_{j,k} \;=\; \sum_{n\in\Z} g_{n-2k}\,a_{j-1,n}.
\end{equation}
\end{Satz}

\begin{proof}
Vgl.\ \cite[Kap.~5.4]{blatter}. Die Reihe in \eqref{eq:zwei-skalen}
konvergiert in $L^2(\R)$, da $(h_n)\in\ell^2(\Z)$ und
$\{\varphi_{j-1,n}\}_{n\in\Z}$ orthonormal ist. Wegen der Stetigkeit des
Skalarprodukts und $h_n\in\R$ folgt
\[
  a_{j,k}
  = \Big\langle f,\, \sum_{n} h_{n-2k}\,\varphi_{j-1,n}\Big\rangle
  = \sum_{n} h_{n-2k}\,\scal{f}{\varphi_{j-1,n}}
  = \sum_{n} h_{n-2k}\,a_{j-1,n}.
\]
Analog für $d_{j,k}$ mit $g$.
\end{proof}

Die Koeffizienten der Skala $j$ sind also gewichtete Summen der
Approximationskoeffizienten der Skala $j-1$; Integrale treten nur noch bei der
Berechnung der Startwerte $a_{0,\cdot}$ auf.

Haben $h$ und $g$ jeweils höchstens $L$ von Null verschiedene Einträge%
%(für die Daubechies-Filter ist $L=2N$)
, so sind die Summen in \eqref{eq:analyse}
endlich: Jeder Koeffizient $a_{j,k}$ bzw.\ $d_{j,k}$ erfordert höchstens $L$
Multiplikationen, vgl.\ \cite[Kap.~5.4]{blatter}. % S. 134
Da die Gewichte mit
$k\mapsto k+1$ um zwei Indizes weiterrücken, halbiert sich zudem die Anzahl
der Koeffizienten von Skala zu Skala bis auf einen Randterm der Größe $L-2$,
vgl.\ \cite[Kap.~5.4, (7)]{blatter}. % S. 133 (Gl. (7)), Längenschranke S. 134


\begin{algorithm}[H]
\caption[Schnelle Wavelet-Transformation]{Schnelle Wavelet-Transformation,
vgl.\ \cite[Kap.~5.4]{blatter}, \cite[Kap.~2.3, Abb.~2.8]{lmr}}
% Blatter S. 132--133; LMR S. 134
\label{alg:mallat}
\begin{algorithmic}[1]
\Require Startfolge $a_{0,\cdot}$, Filter $h$, $g$, Tiefe $J\in\N$
\Ensure Detailfolgen $d_{1,\cdot},\dots,d_{J,\cdot}$ und Approximation $a_{J,\cdot}$
\For{$j = 1,\dots,J$}
  \For{alle $k$}
    \State $a_{j,k} \gets \sum_{n} h_{n-2k}\,a_{j-1,n}$
    \State $d_{j,k} \gets \sum_{n} g_{n-2k}\,a_{j-1,n}$
  \EndFor
\EndFor
\State \Return $d_{1,\cdot},\dots,d_{J,\cdot},\;a_{J,\cdot}$
\end{algorithmic}
\end{algorithm}

\begin{Beispiel}[Haar, Fortsetzung von Beispiel~\ref{bsp:haar-wavelet-koeff}]
\label{bsp:mallat-haar}
Für das Haar-Filter $h_0=h_1=\tfrac{1}{\sqrt2}$ liefert \eqref{eq:g-aus-h}
$g_0=\tfrac{1}{\sqrt2}$, $g_1=-\tfrac{1}{\sqrt2}$; alle übrigen $h_k$, $g_k$
verschwinden. In \eqref{eq:analyse} bleiben daher nur die Summanden
$n=2k$ und $n=2k+1$:
\[
  a_{j,k} = \frac{a_{j-1,2k} + a_{j-1,2k+1}}{\sqrt2},
  \qquad
  d_{j,k} = \frac{a_{j-1,2k} - a_{j-1,2k+1}}{\sqrt2},
\]
also genau Mitteln und Differenzbilden aus Beispiel~\ref{bsp:haar-mittelung}.
Für die Treppenfunktion $f=(4,2,5,7)$ der Haar-MSA ist $a_{0,k} = f(k)$. Die
erste Stufe liefert
$a_1 = (\tfrac{6}{\sqrt2},\tfrac{12}{\sqrt2}) = (3\sqrt2,6\sqrt2)$ und
$d_1 = (\sqrt2,-\sqrt2)$, die zweite $a_{2,0} = 9$ und $d_{2,0} = -3$. Das sind
genau die Koeffizienten aus Beispiel~\ref{bsp:haar-wavelet-koeff}, nun ohne ein
einziges Integral berechnet.
\end{Beispiel}

Der Aufwand ist gering. Hat der Filter die Länge $L$, kostet eine Stufe mit $N$
Eingangswerten etwa $LN$ Multiplikationen. Da sich die Anzahl der Werte von
Stufe zu Stufe ungefähr halbiert, sind es insgesamt etwa
$L\,(N + N/2 + N/4 + \cdots) \approx 2LN$, also $O(N)$ Operationen
\cite[Kap.~5.4]{blatter}. % S. 134--135
Die schnelle Fourier-Transformation braucht dagegen $O(N\log N)$
\cite[§~53]{hankebourgeois}. % S. 405

Jeder Schritt lässt sich exakt rückgängig machen.

\begin{Satz}[Rekonstruktion, {\cite[Kap.~5.4]{blatter}}]
% S. 134--135
\label{satz:rekonstruktion}
Für alle $j,n\in\Z$ gilt
\begin{equation}
\label{eq:synthese}
  a_{j-1,n} \;=\; \sum_{k\in\Z} h_{n-2k}\,a_{j,k} \;+\; \sum_{k\in\Z} g_{n-2k}\,d_{j,k} .
\end{equation}
\end{Satz}

\begin{proof}
Nach Definition~\ref{def:detailraeume} und Korollar~\ref{kor:wavelet-basis} ist
$P_{j-1}f = P_jf + Q_jf = \sum_k a_{j,k}\varphi_{j,k} + \sum_k d_{j,k}\psi_{j,k}$.
Skalarprodukt mit $\varphi_{j-1,n}$ und Lemma~\ref{lem:zwei-skalen} liefern
\eqref{eq:synthese}, vgl.\ \cite[Kap.~5.4]{blatter}. % S. 134--135
\end{proof}

Im Haar-Fall wird daraus
$a_{j-1,2k} = (a_{j,k}+d_{j,k})/\sqrt2$ und
$a_{j-1,2k+1} = (a_{j,k}-d_{j,k})/\sqrt2$, im Beispiel
$(3\sqrt2\pm\sqrt2)/\sqrt2 = 4$ bzw.\ $2$. Auch die Rekonstruktion kostet nur
$O(N)$ Operationen, bleibt also weiterhin linear \cite[Kap.~5.4]{blatter}. % S. 135

\begin{Bemerkung}
\label{bem:mallat-orthogonal}
Jeder Schritt $a_{j-1,\cdot}\mapsto(a_{j,\cdot},d_{j,\cdot})$ von
Algorithmus~\ref{alg:mallat} ersetzt die Orthonormalbasis
$\{\varphi_{j-1,n}\}_n$ von $V_{j-1}$ durch die Orthonormalbasis
$\{\varphi_{j,k}\}_k\cup\{\psi_{j,k}\}_k$ \cite[Kap.~5.4]{blatter}. % S. 134
Die Parseval-Gleichung \cite[Satz~25.18\,(5)]{serov}, angewandt auf
$P_{j-1}f$ in beiden Basen, liefert
\[
  \sum_n \lvert a_{j-1,n}\rvert^2
  \;=\; \norm{P_{j-1}f}^2
  \;=\; \sum_k \lvert a_{j,k}\rvert^2 + \sum_k \lvert d_{j,k}\rvert^2 ,
\]
im Beispiel $4^2+2^2 = 20 = (3\sqrt2)^2 + (\sqrt2)^2$. Ein Schritt ist also eine
orthogonale Abbildung der Koeffizientenfolgen; genau diese Eigenschaft wird in
Abschnitt~\ref{sec:onb-invarianz} gebraucht.
\end{Bemerkung}

In der Anwendung ist $f$ nicht bekannt, sondern nur durch Abtastwerte gegeben.
Diese werden als Koeffizienten $a_{0,k}$ der feinsten Skala verwendet
\cite[Kap.~5.4]{blatter}; % S. 131
wie ein endliches Signal dabei am Rand fortgesetzt wird, beschreibt
Abschnitt~\ref{sec:onb-invarianz}.

\section{Daubechies-Wavelets}
\label{sec:daubechies}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\quellen{Blatter, Kap.~3.5, S.~74 [bestätigt]; LMR, Lemma~2.4.30, S.~176
[bestätigt]; LMR, Satz~2.4.9, S.~153 [bestätigt]; LMR, Kap.~2.4.4,
S.~173--175 [bestätigt], insb.\ Satz~2.4.28 (Momente), S.~174, Träger von
$\psi_N$, S.~175, Regularität $H^{\alpha N}$, $\alpha>0{,}2$, S.~173;
Blatter, Kap.~6.2, S.~146--151 [bestätigt]; Blatter, Kap.~6.3, S.~158
[bestätigt]; Rebstadt, Kap.~2.3.1, S.~16--20 [bestätigt].}

Die Formeln des Mallat-Algorithmus gelten für die Skalierungsfolge jeder MSA;
offen ist aber, welche man wählen soll. Das Haar-Filter erzeugt selbst auf glatten Signalabschnitten von null
verschiedene Detailkoeffizienten. Gesucht
sind daher kurze Filter, deren Wavelet glatte Abschnitte möglichst nicht
sieht. Die dafür entscheidende Eigenschaft ist die folgende.

\begin{Definition}[Verschwindende Momente, vgl.\ {\cite[Kap.~3.5]{blatter}}]
\label{def:momente}
% Blatter S. 74 (dort "Ordnung N" mit M_N != 0)
Ein Wavelet $\psi$ mit kompaktem Träger hat $p$ verschwindende
Momente,\footnote{Wie bei Mallat bezeichnet $p$ die Zahl der Momente; $N$
  bleibt der Signallänge vorbehalten.} falls
\[
  \int_\R t^m\,\psi(t)\,dt \;=\; 0 \qquad\text{für } m = 0,1,\dots,p-1 .
\]
\end{Definition}

Das Moment $m=0$ ist die Zulässigkeitsbedingung aus
Satz~\ref{satz:zulaessigkeit-mittelwert}. Allgemein ist
$\int_\R q(t)\,\psi_{j,k}(t)\,dt = 0$ für jedes Polynom $q$ vom Grad $<p$: Das
Wavelet sieht Polynome niedrigen Grades nicht. Wie sich das im
Mallat-Algorithmus auswirkt, zeigt das folgende Beispiel.

\begin{Beispiel}[Ein lineares Signal]
\label{bsp:momente}
Nach \eqref{eq:analyse} liefert die linear ansteigende Folge $a_{0,n} = n$ die
Detailkoeffizienten
\[
  d_{1,k} \;=\; \sum_m g_m\,(m+2k) \;=\; \sum_m m\,g_m + 2k\sum_m g_m .
\]
Für Haar ist $g_0 = -g_1 = 1/\sqrt2$ (Beispiel~\ref{bsp:haar-g}), also
$\sum_m g_m = 0$, aber $\sum_m m\,g_m = g_1 = -1/\sqrt2$. Damit ist
$d_{1,k} = -1/\sqrt2$ für jedes $k$, obwohl das Signal überall gleichmä\ss{}ig
ansteigt und weder Spitzen noch Knicke besitzt.
Für die Koeffizienten $h_0,\dots,h_3$ aus \eqref{eq:db2-koeff} ist
$(g_{-2},g_{-1},g_0,g_1) = (h_3,-h_2,h_1,-h_0)$, und es gilt
\begin{align*}
  4\sqrt2\,\sum_m g_m &\;=\; (1-\sqrt3)-(3-\sqrt3)+(3+\sqrt3)-(1+\sqrt3) \;=\; 0,\\
  4\sqrt2\,\sum_m m\,g_m &\;=\; -2(1-\sqrt3)+(3-\sqrt3)-(1+\sqrt3) \;=\; 0,
\end{align*}
also $d_{1,k} = 0$: Lineare Abschnitte landen vollständig in den
Approximationskoeffizienten, vgl.\ \cite[Lemma~2.4.29]{lmr}. % S. 174--175: Polynome vom Grad < N liegen in V_0; im Beweis (S. 175 unten): d_k^j = 0
Damit ist auch die Frage aus Beispiel~\ref{bsp:db2-filter} beantwortet: Die
Folge \eqref{eq:db2-koeff} ist das Daubechies-Filter mit zwei verschwindenden
Momenten \cite[Kap.~6.3]{blatter}, % S. 158
das Haar-Filter aus derselben Schar hat nur eines.
\end{Beispiel}
\label{bsp:rampe}

Für glatte, nicht polynomiale Signale gilt das näherungsweise, umso besser,
je feiner die Skala ist.

\begin{Lemma}[Abklingen der Detailkoeffizienten, vgl.\ {\cite[Beweis von Lemma~2.4.30]{lmr}}]
\label{lem:abklingen}
% LMR S. 176 unten (dort mit k statt p, l statt j)
Sei $\psi$ ein Wavelet mit $p$ verschwindenden Momenten und
$\supp\psi = [\alpha,\beta] \ni 0$. Ist $f$ auf $I_{j,k} := \supp\psi_{j,k}$ $p$-mal stetig
differenzierbar mit $\lvert f^{(p)}\rvert \le M_{j,k}$, so gilt
\[
  \lvert d_{j,k}\rvert \;\le\; 2^{j(p+1/2)}\,\frac{M_{j,k}}{p!}\int_\R \lvert t^p\,\psi(t)\rvert\,dt .
\]
\end{Lemma}

\begin{proof}
Die Substitution $x = 2^j(t+k)$ ergibt
$d_{j,k} = 2^{j/2}\int_\R f(2^jk + 2^jt)\,\psi(t)\,dt$. Nach Taylor
\cite[(2.4.20)]{lmr} ist $f(2^jk + 2^jt) = q(t) + R(t)$ mit einem Polynom $q$
vom Grad $<p$ und dem Rest
$R(t) = \tfrac{1}{p!}f^{(p)}(\eta_t)\,(2^jt)^p$, $\eta_t \in I_{j,k}$. Wegen
der verschwindenden Momente ist $\int_\R q\,\psi\,dt = 0$, also
$d_{j,k} = 2^{j/2}\int_\R R(t)\,\psi(t)\,dt$, und
$\lvert R(t)\rvert \le \tfrac{M_{j,k}}{p!}\,2^{jp}\,\lvert t\rvert^p$ liefert
die Behauptung.
\end{proof}

Nach Lemma~\ref{lem:abklingen} werden die Detailkoeffizienten glatter
Abschnitte auf feinen Skalen ($j \to -\infty$) also schnell klein, umso
schneller, je grö\ss{}er $p$ ist. An Sprüngen oder steilen Spitzen ist die
Voraussetzung des Lemmas verletzt, und die Koeffizienten bleiben dort gro\ss{}.

Für den Mallat-Algorithmus braucht man zugleich kurze Filter, denn er ist nur
mit endlichen Filtern exakt ausführbar. Viele verschwindende Momente und kurze

\section{Rauschmodell und diskrete Wavelet-Transformation}
\label{sec:onb-invarianz}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Das Rauschen wird als zufällig modelliert; die dafür benötigten Begriffe
stehen in Abschnitt~\ref{sec:wahrscheinlichkeit}.

\begin{Definition}[Gau\ss'sches wei\ss es Rauschen, {\cite[Kap.~11.2.1--11.2.2]{mallat}}] % S. 548, 552
\label{def:weisses-rauschen}
Ein Zufallsvektor $W = (W_0,\dots,W_{N-1})$ hei\ss t wei\ss es Rauschen der
Varianz $\sigma^2$, wenn $\mathbb E[W_n] = 0$ und
$\mathbb E[W_nW_k] = \sigma^2\,\delta_{n,k}$ für alle $n,k$ gilt. Sind die
$W_n$ zudem gemeinsam normalverteilt, hei\ss t $W$ Gau\ss'sches wei\ss es
Rauschen.
\end{Definition}

Nach Satz~\ref{satz:unkorreliert-unabhaengig} sind die Komponenten eines
Gau\ss'schen wei\ss en Rauschens der Varianz $1$ unabhängig und
standardnormalverteilt.

\paragraph{Das Modell.}
Beobachtet werden $N$ Messwerte
\begin{equation}
\label{eq:rauschmodell}
  Y_n \;=\; f_n + \sigma Z_n, \qquad n = 0,\dots,N-1,
\end{equation}
mit Gau\ss'schem wei\ss em Rauschen $Z$ der Varianz $1$ und Rauschstärke
$\sigma>0$ \cite[Gl.~(11.26)]{mallat}. % S. 548 (dort X = f + W mit W = sigma Z)
Anders als in den Kapiteln~\ref{chapter:mra} und~\ref{chapter:dwt} ist $f$
hier keine Funktion, sondern der unbekannte Vektor $f\in\R^N$ der Abtastwerte
des Signals. Die Messwerte dienen direkt als Startkoeffizienten $a_{0,n}$ von
Algorithmus~\ref{alg:mallat} \cite[Kap.~3.1.1]{lmr}. % S. 237--238 "Wir fassen zusammen"
Das Rauschen wird mit $Z$ bezeichnet, da $W$ bereits für die Detailräume
vergeben ist.

\begin{Definition}[Schätzer und Risiko, {\cite[Kap.~1.2.4]{mallat}}] % S. 12
\label{def:risiko}
Ein Schätzer ordnet den Messwerten $Y$ einen Vektor $\hat f\in\R^N$ zu, der $f$
annähern soll. Sein Risiko ist der erwartete quadratische Fehler
\[
  r(f) \;:=\; \mathbb E\,\norm{\hat f - f}^2 .
\]
Für eine einzelne Realisierung des Rauschens hei\ss t $\norm{\hat f - f}^2$ die
Fehlerenergie.
\end{Definition}

Ohne jede Bearbeitung, also mit $\hat f = Y$, ist die Fehlerenergie die
Energie des Rauschens $\norm{\sigma Z}^2$, und ihr Erwartungswert ist
$r(f) = N\sigma^2$ \cite[Kap.~11.1.1]{mallat}. % S. 544: "total noise energy E{||W||^2} = N sigma^2"

\paragraph{Die DWT eines endlichen Signals.}
In Algorithmus~\ref{alg:mallat} ist
\[
  a_{1,k} \;=\; \sum_{n\in\Z} h_{n-2k}\,a_{0,n},
\]
die Summe läuft also formal über alle ganzen Zahlen $n$. Beim Filter
\eqref{eq:db2-koeff} sind nur $h_0,\dots,h_3$ von null verschieden; für jedes
$k$ bleiben daher nur die vier Summanden mit $n = 2k,\dots,2k+3$. Ein
Messsignal hat aber nur die $N$ Werte $a_{0,0},\dots,a_{0,N-1}$. Für $N=8$ ist
etwa
\[
  a_{1,3} \;=\; h_0a_{0,6} + h_1a_{0,7} + h_2a_{0,8} + h_3a_{0,9},
\]
und $a_{0,8}$, $a_{0,9}$ gibt es nicht. Man setzt das Signal daher periodisch
fort, $a_{0,8} := a_{0,0}$, $a_{0,9} := a_{0,1}$ \cite[Kap.~2.4.7.2]{lmr}. % S. 206
Das ist der einfachste von drei Wegen, die Mallat beschreibt; die anderen
spiegeln das Signal am Rand oder verwenden eigene Randwavelets
\cite[Kap.~7.3.1, 7.5]{mallat}. % S. 301--302 ("Finite Signals"), Kap. 7.5 ab S. 317

Schreibt man die vier Gleichungen für $a_{1,0},\dots,a_{1,3}$ untereinander,
entsteht eine Matrix $H\in\R^{4\times8}$ mit $(a_{1,k})_k = H\,(a_{0,n})_n$:
\[
  H \;=\; \begin{pmatrix}
    h_0 & h_1 & h_2 & h_3 & 0 & 0 & 0 & 0\\
    0 & 0 & h_0 & h_1 & h_2 & h_3 & 0 & 0\\
    0 & 0 & 0 & 0 & h_0 & h_1 & h_2 & h_3\\
    h_2 & h_3 & 0 & 0 & 0 & 0 & h_0 & h_1
  \end{pmatrix}.
\]
Jede Zeile ist die vorige, um zwei Stellen verschoben; was rechts herausfällt,
kommt links wieder hinein. Ebenso liefert eine Matrix $G\in\R^{4\times8}$ aus
den $g_k$ die Details $d_{1,0},\dots,d_{1,3}$.

Allgemein sei $M$ gerade und $w\in\R^M$ periodisch fortgesetzt,
$w_{n+M} := w_n$. Die periodischen Mallat-Transformationen
$H_M, G_M\colon\R^M\to\R^{M/2}$ sind
\[
  (H_Mw)_k \;:=\; \sum_{n\in\Z} h_{n-2k}\,w_n, \qquad
  (G_Mw)_k \;:=\; \sum_{n\in\Z} g_{n-2k}\,w_n, \qquad 0\le k<\tfrac M2
\]
\cite[Kap.~3.6.2]{lmr}; % S. 289, dort mit l = n - 2k geschrieben
das sind die Formeln aus Algorithmus~\ref{alg:mallat} mit periodisch gelesenen
Indizes. Wir setzen $S_M := \binom{H_M}{G_M}\in\R^{M\times M}$; das Beispiel
oben ist $H = H_8$.

\begin{Lemma}[{\cite[Kap.~3.6.2, Gl.~(3.6.18)--(3.6.20)]{lmr}}] % S. 289
\label{lem:dwt-orthogonal}
% PRÜFEN: setzt LMR M >= Filterlänge voraus? Für db2 und M = 2 (S_2 im Beispiel
% N = 8, J = 3) gilt die Aussage trotzdem, da h_0+h_2 = h_1+h_3 = 1/sqrt(2).
Für die Filter $h$, $g$ aus Kapitel~\ref{chapter:dwt} gilt $H_MH_M^\top = G_MG_M^\top = I$, $H_MG_M^\top = 0$ und
$H_M^\top H_M + G_M^\top G_M = I$. Die Matrix $S_M$ ist also orthogonal, das
hei\ss t $S_M^\top S_M = S_MS_M^\top = I$.
\end{Lemma}

Der Preis des Periodisierens: Sind Anfang und Ende des Signals verschieden, so
verhalten sich die Koeffizienten am Rand, als hätte das Signal dort einen
Sprung, und werden gro\ss{} \cite[Kap.~7.5.1]{mallat}. % S. 319

\paragraph{Die DWT als orthogonale Matrix.}
Die DWT der Tiefe $J$ wendet $S_N$ auf alle $N$ Werte an und danach jeweils
nur auf die zuletzt berechneten Approximationskoeffizienten; die Details bleiben
stehen. Da jeder Schritt die Anzahl der Approximationskoeffizienten halbiert,
sei $N$ durch $2^J$ teilbar. Als Matrix ist das
\[
  A \;:=\; \begin{pmatrix} S_{N/2^{J-1}} & 0\\ 0 & I\end{pmatrix}\cdots
           \begin{pmatrix} S_{N/2} & 0\\ 0 & I\end{pmatrix} S_N
  \;\in\;\R^{N\times N},
  \qquad AY = (a_J, d_J, \dots, d_1),
\]
für $N = 8$, $J = 3$ etwa
\[
  A = \begin{pmatrix} S_2 & 0\\ 0 & I_6\end{pmatrix}
      \begin{pmatrix} S_4 & 0\\ 0 & I_4\end{pmatrix} S_8,
  \qquad AY = (a_3, d_3, d_2, d_1)
\]
mit $1, 1, 2$ und $4$ Einträgen.
Jeder Faktor ist nach Lemma~\ref{lem:dwt-orthogonal} orthogonal, also auch $A$.
Wir bezeichnen die Zeilen von $A$ mit $v_0,\dots,v_{N-1}$. Sie bilden eine
Orthonormalbasis des $\R^N$, und jeder Koeffizient ist ein Skalarprodukt mit
einer von ihnen; Mallat nennt sie diskrete Wavelets und Skalierungsfunktionen
\cite[Kap.~7.5, Gl.~(7.173)]{mallat}. % S. 318 (dort phi^int_{j,n}, psi^int_{j,n} statt v_m); in Kap. 11 g_m
Aus \eqref{eq:rauschmodell} wird damit
\begin{equation}
\label{eq:rauschmodell-koeff}
  \tilde Y_m \;=\; \theta_m + \sigma\tilde Z_m \qquad\text{mit }
  \tilde Y_m = \scal{Y}{v_m},\ \theta_m = \scal{f}{v_m},\ \tilde Z_m = \scal{Z}{v_m},
\end{equation}
kurz $\tilde Y = AY$, $\theta = Af$, $\tilde Z = AZ$ \cite[Kap.~11.2.1]{mallat}. % S. 548, dort X_B = f_B + W_B mit g_m statt v_m

\paragraph{Was mit dem Rauschen geschieht.}

\begin{Lemma}[{\cite[Kap.~11.2.1--11.2.2]{mallat}}] % S. 549, 554, 556
\label{lem:rauschen-invariant}
$\tilde Z$ ist wieder Gau\ss'sches wei\ss es Rauschen der Varianz $1$; die
$\tilde Z_0,\dots,\tilde Z_{N-1}$ sind also unabhängig und standardnormalverteilt.
\end{Lemma}

\begin{proof}
Wegen der Linearität des Erwartungswerts ist
$\mathbb E[\tilde Z_m] = \sum_n v_m[n]\,\mathbb E[Z_n] = 0$ und
\[
  \mathbb E\bigl[\tilde Z_m\tilde Z_p\bigr]
  = \sum_{n=0}^{N-1}\sum_{k=0}^{N-1} v_m[n]\,v_p[k]\,\mathbb E[Z_nZ_k]
  = \sum_{n=0}^{N-1} v_m[n]\,v_p[n]
  = \scal{v_m}{v_p} = \delta_{m,p}
\]
\cite[Kap.~11.2.1]{mallat}, % S. 549
da die $v_m$ eine Orthonormalbasis bilden. Die $\tilde Z_m$ sind
Linearkombinationen der gemeinsam normalverteilten $Z_n$ und damit selbst
gemeinsam normalverteilt, vgl.\ \cite[Kap.~11.2.2]{mallat}; % S. 554, 556: dort nur das Ergebnis
nach Satz~\ref{satz:unkorreliert-unabhaengig} sind sie also unabhängig.
\end{proof}

Die Transformation verändert also das Signal, die Verteilung des Rauschens
aber nicht: Jeder Koeffizient trägt Rauschen derselben Stärke $\sigma$, und das
Rauschen beeinflusst die Wahl der Basis nicht \cite[Kap.~11.2.1]{mallat}. % S. 549

\begin{Beispiel}[Ein verrauschter Sinus]
\label{bsp:sinus}

\[

\]
Jede Periode umfasst also $256$ Abtastwerte, insgesamt sind es acht volle
Perioden. Dazu kommt Rauschen der Stärke $\sigma = 0{,}3$: Die $Z_n$ werden als

\section{Diagonale Schätzer und Thresholding}
\label{sec:duennbesetztheit}
\label{sec:thresholding}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Nach Abschnitt~\ref{sec:onb-invarianz} bilden die Zeilen $v_0,\dots,v_{N-1}$
von $A$ eine Orthonormalbasis des $\R^N$, und die Koeffizienten der Messwerte
in dieser Basis sind $\tilde Y_m = \scal{Y}{v_m} = \theta_m + \sigma\tilde Z_m$
nach \eqref{eq:rauschmodell-koeff}. Im mittleren Schritt von
Abbildung~\ref{fig:dreischritt} wird jeder dieser Koeffizienten für sich mit
einem Faktor $\lambda_m$ multipliziert, der angibt, wie viel von ihm übrig
bleibt: $\lambda_m = 1$ behält ihn, $\lambda_m = 0$ löscht ihn, ein Wert
dazwischen verkleinert ihn.

\begin{Definition}[Diagonaler Schätzer, {\cite[Kap.~11.2.1, Gl.~(11.27)]{mallat}}] % S. 549; dort a_m(X_B[m]) statt lambda_m, g_m statt v_m
\label{def:diagonal}
Ein Schätzer hei\ss t diagonal in der Basis $\{v_m\}$, wenn er jeden
Koeffizienten mit einem Faktor $\lambda_m$ multipliziert,
\[
  \hat\theta_m \;=\; \lambda_m\,\tilde Y_m \qquad (m = 0,\dots,N-1),
\]
und das Ergebnis mit $A^{-1} = A^\top$ zurücktransformiert:
$\hat f = A^\top\hat\theta$. Dabei darf $\lambda_m$ von den Messwerten nur über
$\tilde Y_m$ abhängen, nicht über die übrigen Koeffizienten $\tilde Y_k$,
$k\neq m$. Hängt kein $\lambda_m$ von den Messwerten ab, hei\ss t der Schätzer
linear.
\end{Definition} 

In den Koeffizienten ist ein diagonaler Schätzer die Multiplikation mit einer
Diagonalmatrix,
\[
  \hat\theta \;=\;
  \begin{pmatrix}
    \lambda_0 & 0 & \cdots & 0 \\
    0 & \lambda_1 & \ddots & \vdots \\
    \vdots & \ddots & \ddots & 0 \\
    0 & \cdots & 0 & \lambda_{N-1}
  \end{pmatrix}
  \tilde Y .
\]
Die beiden wichtigsten diagonalen Schätzer sind nicht linear: Sie vergleichen
$\lvert\tilde Y_m\rvert$ mit einer Schwelle $T>0$, die Faktoren hängen also von
$\tilde Y_m$ selbst ab \cite[Kap.~11.2.2]{mallat}. % S. 552 (11.45); S. 553 "nonlinear projector"

\begin{Definition}[Hard- und Soft-Thresholding, {\cite[Gl.~(11.47), (11.49)]{mallat}}] % S. 552--553
\label{def:thresholding}
Für $T>0$ hei\ss en
\[
  \rho^{\mathrm H}_T(x) :=
  \begin{cases} x, & \lvert x\rvert > T,\\ 0, & \lvert x\rvert\le T,\end{cases}
  \qquad
  \rho^{\mathrm S}_T(x) :=
  \begin{cases} x - T, & x > T,\\ x + T, & x < -T,\\ 0, & \lvert x\rvert\le T\end{cases}
\]
Hard- bzw.\ Soft-Thresholding. Der zugehörige Schätzer berechnet mit $\rho_T$
für eine der beiden Funktionen die Koeffizienten
$\hat\theta_m = \rho_T(\tilde Y_m)$ und transformiert zurück:
$\hat f = A^\top\hat\theta$.
\end{Definition}

Mittelwert von $4{,}5$ auf $3{,}5$; darauf kommt Abschnitt~\ref{sec:mad}
zurück. In beiden Fällen wird jeder Koeffizient allein
anhand seines eigenen Betrags behandelt (Abbildung~\ref{fig:thresholding}).
\end{Beispiel}

Welches Risiko ein diagonaler Schätzer bestenfalls erreicht, lässt sich
koeffizientenweise untersuchen: Das Risiko zerfällt in eine Summe von
Einzelbeiträgen.

\begin{Lemma}[{\cite[Gl.~(11.28)]{mallat}}] % S. 549
\label{lem:risiko-summe}
Für jeden Schätzer der Form $\hat f = A^\top\hat\theta$ gilt
\begin{equation}
\label{eq:risiko-summe}
  \mathbb E\,\norm{\hat f - f}^2 \;=\; \sum_{m=0}^{N-1}\mathbb E\,\lvert\hat\theta_m - \theta_m\rvert^2 .
\end{equation}
\end{Lemma}

%\begin{proof}
%Hier ist $\norm{x}^2 = \sum_{n=0}^{N-1} x_n^2$ die euklidische Norm des $\R^N$.
%Wegen $\hat f = A^\top\hat\theta$ und $A A^\top = I$ ist $A\hat f = \hat\theta$,
%und nach \eqref{eq:rauschmodell-koeff} ist $Af = \theta$. Da $A$ nach
%Lemma~\ref{lem:dwt-orthogonal} orthogonal ist und daher die Länge erhält, folgt
%\[
%  \norm{\hat f - f}^2 \;=\; \norm{A(\hat f - f)}^2 \;=\; \norm{\hat\theta - \theta}^2
%  \;=\; \sum_{m=0}^{N-1}\lvert\hat\theta_m - \theta_m\rvert^2 .
%\]
%Bilden des Erwartungswerts, der linear ist, liefert \eqref{eq:risiko-summe}.
%\end{proof}

Bei einem diagonalen Schätzer ist $\hat\theta_m = \lambda_m\tilde Y_m$; der
$m$-te Summand in \eqref{eq:risiko-summe} hängt also nur von $\lambda_m$ ab, und
das Risiko wird minimiert, indem man jeden Summanden einzeln minimiert. Bekannt
ist dabei nur die Summe $\tilde Y_m = \theta_m + \sigma\tilde Z_m$, nicht ihre
beiden Anteile. Wir betrachten zunächst feste Faktoren $\lambda_m \in \{0,1\}$,
also diagonale Schätzer, die jeden Koeffizienten entweder behalten oder
löschen \cite[Kap.~11.2.1]{mallat}: % S. 550

\begin{itemize}
\item Behalten, $\lambda_m = 1$: Das Signal bleibt, aber auch das Rauschen; die
  Abweichung ist $\hat\theta_m - \theta_m = \sigma\tilde Z_m$. Da $\tilde Z_m$
  nach Lemma~\ref{lem:rauschen-invariant} Mittelwert $0$ und Varianz $1$ hat,
  ist $\mathbb E\lvert\hat\theta_m - \theta_m\rvert^2 = \sigma^2\,\mathbb E[\tilde Z_m^2] = \sigma^2$.
\item Löschen, $\lambda_m = 0$: Dann ist $\hat\theta_m = 0$. Das Rauschen ist
  weg, aber ebenso das Signal; die Abweichung ist $-\theta_m$, der Fehler
  $\theta_m^2$.
\end{itemize}
Behalten kostet also im Mittel $\sigma^2$, Löschen $\theta_m^2$. Behalten ist
daher genau dann mindestens so gut wie Löschen, wenn $\lvert\theta_m\rvert$
mindestens so gro\ss{} wie $\sigma$ ist: Für
$\sigma = 0{,}3$ ist etwa bei $\theta_m = 2$ Behalten besser ($0{,}09$ statt $4$),
bei $\theta_m = 0{,}1$ Löschen ($0{,}01$ statt $0{,}09$).

\begin{Definition}[Ideale Auswahl, {\cite[Gl.~(11.32)--(11.34)]{mallat}}] % S. 550 "oracle projector", dort r_pr statt r_id; S. 556 "ideal coefficient selection"
\label{def:ideale-auswahl}
Die ideale Auswahl ist der diagonale Schätzer mit den festen Faktoren
$\lambda_m = 1$ für $\lvert\theta_m\rvert \ge \sigma$ und $\lambda_m = 0$ sonst.
\end{Definition}

Nach Lemma~\ref{lem:risiko-summe} ist ihr Risiko
\begin{equation}
\label{eq:orakel}
  r_{\mathrm{id}}(f) \;=\; \sum_{m=0}^{N-1}\min\bigl(\theta_m^2,\,\sigma^2\bigr),
\end{equation}
und kein diagonaler Schätzer mit festen Faktoren $\lambda_m \in \{0,1\}$ hat
ein kleineres Risiko. % Mallat S. 550 "The nonlinear projector that minimizes the risk (11.29)"

Da die $\lambda_m$ der idealen Auswahl vom unbekannten $\theta_m$ abhängen, lässt
sie sich nicht berechnen; Mallat spricht von einem Orakel
\cite[Kap.~11.2.1]{mallat}. % S. 550
Hard-Thresholding ist ihr praktischer Ersatz: Es behält oder löscht ebenfalls
jeden Koeffizienten, entscheidet aber nach dem bekannten $\lvert\tilde Y_m\rvert$
und der Schwelle $T$ statt nach $\lvert\theta_m\rvert$ und $\sigma$. Sein Risiko
ist mindestens $r_{\mathrm{id}}(f)$ \cite[Kap.~11.2.2]{mallat}; % S. 553 oben, unnummerierte Ungleichung
die ideale Auswahl ist also eine untere Schranke für das, was
Hard-Thresholding erreichen kann.

Sind $M$ Koeffizienten mindestens so gro\ss{} wie $\sigma$, so lässt sich das
Risiko der idealen Auswahl schreiben als
\[
  r_{\mathrm{id}}(f) \;=\; \sum_{\lvert\theta_m\rvert<\sigma}\theta_m^2 \;+\; M\sigma^2
\]
\cite[Gl.~(11.36), (11.37)]{mallat}. % S. 550--551; dort r_pr statt r_id, f_B[m] statt theta_m
Der erste Term ist der Fehler, wenn man $f$ nur durch diese $M$ Koeffizienten
darstellt, der zweite das Rauschen auf ihnen. Das Risiko ist also klein, wenn
sich $f$ durch wenige Basisvektoren gut darstellen lässt
\cite[Kap.~11.2.2]{mallat}. % S. 555--556: "f must be well approximated by a few vectors in B"
Wir nennen $f$ dann dünnbesetzt in der Basis $\{v_m\}$
\cite[Kap.~11.2.1]{mallat}. % S. 552 oben: "sparse ... representation with only a few vectors in B"; vgl. S. 548, S. 562

Für Wavelets gilt das bei stückweise glatten Signalen: Gro\ss e Koeffizienten
entstehen nur an scharfen Übergängen, und davon gibt es wenige
\cite[Kap.~11.3.1]{mallat}. % S. 565

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Die Wahl der Schwelle}
\label{sec:universelle-schwelle}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Die Schwelle $T$ soll das Rauschen entfernen, aber möglichst wenig vom Signal.
Ist $f = 0$, so besteht jeder Koeffizient nur aus Rauschen,
$\tilde Y_m = \sigma\tilde Z_m$; damit alle gelöscht werden, muss $T$ grö\ss er
sein als der grö\ss te Rauschanteil $\max_m\lvert\sigma\tilde Z_m\rvert$. Ist
$f\neq0$, darf $T$ nicht viel grö\ss er sein, sonst gehen auch Koeffizienten mit
gro\ss em Signalanteil verloren. $T$ soll also knapp über dem grö\ss ten
Rauschanteil liegen \cite[Kap.~11.2.2]{mallat}. % S. 555--556, "Choice of Threshold"
Dieses Maximum ist unbekannt. Für $N$ unabhängige Rauschanteile liegt es aber
mit hoher Wahrscheinlichkeit knapp unter
\begin{equation}
\label{eq:universelle-schwelle}
  T \;=\; \sigma\sqrt{2\ln N}
\end{equation}
\cite[Gl.~(11.64)]{mallat}. % S. 556
Diese Wahl hei\ss t universelle Schwelle \cite[Kap.~11.3.2]{mallat}. % S. 568: "the universal threshold of Theorem 11.7"
Dass $T$ mit $N$ wächst, liegt an den Ausläufern der Normalverteilung: Unter
mehr Werten kommen auch grö\ss ere vor \cite[Kap.~11.2.2]{mallat}. % S. 556

\begin{Satz}[Donoho--Johnstone, {\cite[Satz~11.7]{mallat}}] % S. 553
\label{satz:donoho-johnstone}
Für $N\ge4$ und $T = \sigma\sqrt{2\ln N}$ erfüllt das Risiko des Hard- und des
Soft-Thresholding-Schätzers
\begin{equation}
\label{eq:donoho-johnstone}
  r(f) \;\le\; (2\ln N + 1)\bigl(\sigma^2 + r_{\mathrm{id}}(f)\bigr).
\end{equation}
\end{Satz}

Für den Beweis siehe \cite[Satz~11.7, Lemma~11.1]{mallat}. % S. 553--555
Zusammen mit der unteren Schranke aus Abschnitt~\ref{sec:thresholding}, die
Mallat für Hard-Thresholding zeigt, ergibt sich dafür
\[
  r_{\mathrm{id}}(f) \;\le\; r(f) \;\le\; (2\ln N + 1)\bigl(\sigma^2 + r_{\mathrm{id}}(f)\bigr)
\]
\cite[Kap.~11.2.2]{mallat}. % S. 553: r_th >= r_pr (unnummeriert) und (11.51)
Bis auf den Faktor $2\ln N + 1$ ist Hard-Thresholding also so gut wie die ideale
Auswahl. Die obere Schranke gilt für beide Varianten: Ist $f$ dünnbesetzt und
damit $r_{\mathrm{id}}(f)$ klein, so ist auch $r(f)$ klein. Mit diagonalen
Schätzern lässt sich dieser Faktor für gro\ss e $N$ nicht verbessern
\cite[Gl.~(11.52)]{mallat}. % S. 554

\section{Schätzung der Rauschstärke}
\label{sec:mad}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Die Schwelle enthält das unbekannte $\sigma$. Mallat schätzt es aus den $N/2$
Detailkoeffizienten $d_{1,k}$ der feinsten Stufe \cite[Kap.~11.3.1]{mallat}: % S. 565
Wo das Signal auf dem Träger des Wavelets glatt ist, ist sein Koeffizient
klein, und $d_{1,k}$ besteht praktisch nur aus Rauschen. Nur an den wenigen
scharfen Übergängen ist er gro\ss; diese Werte nennt Mallat Ausrei\ss er. Auch
beim gemessenen EKG, das in Kapitel \ref{chapter:anwendung} relevant wird, enthält $d_1$ fast das gesamte Rauschen
\cite[Kap.~3.1.2]{lmr}. % S. 239

Der Median von $P$ Zahlen ist der mittlere Wert, wenn man sie der Grö\ss e nach
ordnet, also der Wert vom Rang $P/2$ \cite[Kap.~11.3.1]{mallat}. % S. 565
Anders als der Mittelwert hängt er nicht davon ab, wie gro\ss{} die Werte über ihm
sind, und wird von wenigen Ausrei\ss ern daher kaum verändert. Ist $M$ der
Median der Beträge von $P$ unabhängigen normalverteilten Zufallsvariablen mit
Erwartungswert $0$ und Varianz $\sigma^2$, so ist
$\mathbb E[M]\approx0{,}6745\,\sigma$ \cite[Kap.~11.3.1]{mallat}. % S. 565
Daraus ergibt sich der Schätzer
\begin{equation}
\label{eq:mad}
  \hat\sigma \;=\; \frac{\operatorname{Median}\bigl(\lvert d_{1,k}\rvert : 0\le k<N/2\bigr)}{0{,}6745}
\end{equation}
\cite[Gl.~(11.85)]{mallat}. % S. 565
Er setzt voraus, dass das Rauschen in allen Koeffizienten dieselbe Stärke hat,
was nach Lemma~\ref{lem:rauschen-invariant} für wei\ss es Rauschen zutrifft. Ist
das Rauschen nicht wei\ss, hängt seine Varianz vom Basisvektor ab, und die
Schwelle muss mit dem Koeffizienten variieren
\cite[Kap.~11.2.2]{mallat}; % S. 557, "Colored Noise"
darauf kommt Kapitel~\ref{chapter:anwendung} zurück.

\anmerkung{Kap.~6 greift farbiges Rauschen bisher nicht auf -- ergänzen oder Verweis streichen.}

\paragraph{Das vollständige Verfahren.}
Mallat schwellt alle Koeffizienten der Waveletbasis, auch die der gröbsten
Approximation \cite[Gl.~(11.84)]{mallat}, % S. 563
und bestimmt $\sigma$ mit \eqref{eq:mad} \cite[Kap.~11.3.1]{mallat}. % S. 565
Zusammen ergibt sich:

\begin{algorithm}[H]
\caption[Entrauschen mit universeller Schwelle]{Entrauschen mit universeller
Schwelle nach \cite[Kap.~11.2--11.3]{mallat}}
\label{alg:visushrink}
\begin{algorithmic}[1]
\Require Messwerte $Y\in\R^N$, Filter $h$, $g$, Tiefe $J$ mit $2^{J}\mid N$
\State $(a_J,d_J,\dots,d_1) \gets AY$ \Comment{Algorithmus~\ref{alg:mallat}, periodisiert}
\State $T \gets \hat\sigma\sqrt{2\ln N}$ mit $\hat\sigma$ aus \eqref{eq:mad}
\State ersetze jeden Koeffizienten $c$ von $a_J,d_J,\dots,d_1$ durch $\rho_T(c)$
\State \Return $\hat f \gets A^\top(\text{geschwellte Koeffizienten})$ \Comment{Satz~\ref{satz:rekonstruktion}}
\end{algorithmic}
\end{algorithm}

Satz~\ref{satz:donoho-johnstone} setzt das wahre $\sigma$ voraus; mit dem
geschätzten $\hat\sigma$ ist die Schranke \eqref{eq:donoho-johnstone} nur ein
Anhaltspunkt. Ferner werden in Zeile~3 auch die Approximationskoeffizienten $a_J$
geschwellt. Sie tragen den glatten, groben Anteil des Signals und sind meist
gro\ss; Soft-Thresholding verkleinert jeden von ihnen um $T$ und verschiebt so den
glatten Anteil, wie schon in Beispiel~\ref{bsp:diagonal} den Mittelwert. In
seinem Beispiel zur Bildentrauschung schwellt Mallat die
Skalierungskoeffizienten deshalb nicht \cite[Kap.~11.3.2]{mallat}. % S. 568: "low-frequency scaling coefficients are not thresholded"

\section{Skalen als Frequenzbänder}
\label{sec:frequenzbaender}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Um zu entscheiden, welche Stufen der DWT gelöscht, geschwellt oder
unverändert gelassen werden, ordnen wir jeder Stufe ein Frequenzband zu.
Castillo et al.\ wählen so die Tiefe für die Drift und die Zahl der
geschwellten Stufen \cite[Abschn.~3.1.2, 3.1.3]{castillo}. % S. 4--5

\paragraph{Höchste Frequenz.}
Die Frequenzbänder der Stufen entstehen, indem das Band des Signals von oben
her fortlaufend halbiert wird. Dazu brauchen wir seine obere Grenze. Nach dem Abtasttheorem von Shannon
(vgl.\ \cite[Kap.~2.4, Satz (2.16)]{blatter}) % S. 47--51; dort Kreisfrequenz Omega = pi/T, in Hz: f_s/2
ist ein mit der Rate $f_s$ abgetastetes Signal nur bis zur Frequenz
\[
  F_{\max} = \frac{f_s}{2}
\]
eindeutig darstellbar \cite[Abschn.~3.1.2]{castillo}; % S. 5
eine Schwingung der Frequenz $f > F_{\max}$ hat an den Abtastzeiten
$n/f_s$ dieselben Werte wie eine der Frequenz $f_s - f < F_{\max}$.
In der Variablen $\omega = 2\pi f/f_s$ der Filter aus
Kapitel~\ref{chapter:dwt} entspricht $[0,F_{\max}]$ dem Intervall $[0,\pi]$.
Damit liegt das gesamte Signal $Y$ im Band $[0,F_{\max}]$; dieses Band
verteilt die DWT im Folgenden auf die Stufen.

\paragraph{Halbierung des Bandes.}
Die Fourier-Filter
\[
  H(\omega) = \frac{1}{\sqrt2}\sum_k h_k e^{-ik\omega}, \qquad
  G(\omega) = \frac{1}{\sqrt2}\sum_k g_k e^{-ik\omega}
\]
aus \eqref{eq:symbol-H} und \eqref{eq:G-aus-H} sind nach
Abschnitt~\ref{sec:fourier-bedingungen} ein Tief- und ein Hochpass:
$\lvert H\rvert$ ist auf $[0,\pi/2]$ gro\ss{} und auf $[\pi/2,\pi]$ klein,
$\lvert G\rvert$ umgekehrt (Abbildung~\ref{fig:tief-hochpass}).
\anmerkung{Abbildung~\ref{fig:tief-hochpass} zeigt nur Haar und db2, nicht
das hier verwendete $\psi_6$. Empfehlung: beim Beispiel~\ref{bsp:db6} in
Kapitel~\ref{chapter:dwt} eine Teilabbildung mit $\lvert H\rvert$ und
$\lvert G\rvert$ für $\psi_6$ ergänzen und hier darauf verweisen. Werte zum
Vergleich: $\lvert H(\pi/4)\rvert = 0{,}924$ / $0{,}971$ / $0{,}999$ und
$\lvert H(3\pi/4)\rvert = 0{,}383$ / $0{,}241$ / $0{,}048$ für Haar / db2 /
db6.}
Eine Stufe der DWT wendet nach \eqref{eq:analyse} beide auf $a_{i-1}$ an und
behält jeden zweiten Wert: $a_i$ entsteht durch den Tiefpass und erhält die
untere Hälfte des Bandes von $a_{i-1}$, $d_i$ entsteht durch den Hochpass und
erhält die obere \cite[Abschn.~3.1.2]{castillo}. % S. 4--5
Mit $a_0 = Y$ und dem Band $[0, F_{\max}]$ von $Y$ folgt
\begin{equation}
\label{eq:frequenzbaender}
  a_i \;\leftrightarrow\; \Bigl[0,\,\frac{F_{\max}}{2^{i}}\Bigr],
  \qquad
  d_i \;\leftrightarrow\; \Bigl[\frac{F_{\max}}{2^{i}},\,\frac{F_{\max}}{2^{i-1}}\Bigr]
\end{equation}
\cite[Kap.~2.3.2, Tab.~2.4]{rebstadt}. % S. 27--28
Bei Castillo et al.\ ist $f_s = 250$\,Hz, also $F_{\max} = 125$\,Hz, und $a_7$
erfasst $0$ bis $125/2^7 \approx 0{,}977$\,Hz \cite[Abschn.~3.1.2]{castillo}. % S. 5
Die Zuordnung ist nicht scharf, da $\lvert H\rvert$ und $\lvert G\rvert$
nicht sprunghaft von $1$ auf $0$ fallen. Wegen
$\lvert H\rvert^2 + \lvert G\rvert^2 = 1$
(Abschnitt~\ref{sec:fourier-bedingungen}) und
$\lvert G(\pi/2)\rvert = \lvert H(\pi/2)\rvert$ nach \eqref{eq:G-aus-H} gilt
an der Grenze $\lvert H(\pi/2)\rvert = \lvert G(\pi/2)\rvert = 1/\sqrt2$:
Eine Frequenz genau auf einer Bandgrenze verteilt ihre Energie je zur Hälfte
auf beide benachbarten Stufen, Frequenzen nahe der Grenze auf beide.

\section{Das Verfahren}
\label{sec:verfahren}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Wir übernehmen das Ein-Schritt-Verfahren von Castillo et al., das Drift und
Rauschen mit einer einzigen Zerlegung und Rekonstruktion entfernt
\cite[Abschn.~3]{castillo}. % S. 4
Abschnitt~\ref{sec:ein-schritt} definiert das Verfahren, die folgenden
Abschnitte wählen seine Parameter.

\subsection{Modell und Ein-Schritt-Verfahren}
\label{sec:ein-schritt}

Castillo et al.\ und Alessio gehen für das Rauschen vom Modell
\eqref{eq:rauschmodell} aus, $Y_n = f_n + \sigma Z_n$
\cite[Abschn.~2.2.2, Gl.~(8)]{castillo}, % S. 3
\cite[Kap.~15.2.1]{alessio}, % S. 717
und behandeln die Basisliniendrift getrennt davon \cite[Abschn.~2.2.1]{castillo}. % S. 3
Wir fassen beides in einer Gleichung zusammen; diese Schreibweise ist eigen:
\begin{equation}
\label{eq:modell-ekg}
  Y_n = f_n + b_n + \sigma Z_n, \qquad n = 0,\dots,N-1 .
\end{equation}
Dabei ist $f\in\R^N$ das EKG, $b\in\R^N$ die Basisliniendrift und
$\sigma Z\in\R^N$ wie in \eqref{eq:rauschmodell} Gau\ss'sches wei\ss es
Rauschen.
Elektrodenbewegung, Netzbrummen und Muskelrauschen aus
Tabelle~\ref{tab:stoerungen} sind darin nicht enthalten.
Es bezeichne $A$ die periodisierte DWT der Tiefe $J$ aus
Abschnitt~\ref{sec:onb-invarianz} mit dem Filter \eqref{eq:db6-koeff} des
Wavelets $\psi_6$ (Beispiel~\ref{bsp:db6}). $AY$ ist die Zerlegung des
Signals in die Stufen, also die Approximation $a_J$ und die Details
$d_J, \dots, d_1$:
\[
  AY = (a_J, d_J, d_{J-1}, \dots, d_1) \in \R^N, \qquad
  a_J \in \R^{N/2^J},\; d_j \in \R^{N/2^j}.
\]

\begin{Bemerkung}[Wahl des Wavelets]
Castillo et al.\ wählen das Wavelet nach seiner Ähnlichkeit mit dem
QRS-Komplex und nehmen \texttt{db6}, also $\psi_6$ aus
Satz~\ref{satz:daubechies} mit $p = 6$ verschwindenden Momenten, weil es
einem echten EKG ähnelt \cite[Abschn.~3.1.1]{castillo}. % S. 4
Allgemein soll das Wavelet die Form der Merkmale haben, die erhalten bleiben
sollen \cite[Kap.~15.2.1]{alessio}. % S. 719
In ihrem Vergleich von zwölf Wavelets an synthetischen EKG finden Castillo
et al.\ zwischen den Wavelets keine gro\ss en Unterschiede
\cite[Abschn.~4, Tab.~2]{castillo}. % S. 8, 12
\end{Bemerkung}

Da $A\colon\R^N\to\R^N$ linear ist, folgt aus \eqref{eq:modell-ekg}
\begin{equation}
\label{eq:modell-koeff}
  AY = Af + Ab + \sigma AZ .
\end{equation}
Nach Lemma~\ref{lem:rauschen-invariant} hat jeder Koeffizient von $\sigma AZ$
die Varianz $\sigma^2$, gleich in welcher Stufe er liegt; $Ab$ liegt bei
geeigneter Wahl von $J$ nach \eqref{eq:frequenzbaender} näherungsweise nur in
$a_J$, für $J = 8$ nach Tabelle~\ref{tab:baender} in $a_8$ und am Rand von
$d_8$.

Castillo et al.\ gehen in vier Schritten vor \cite[Abschn.~3]{castillo}; % S. 4, Schritte (1)--(4)
sie schreiben dabei $L$ für unser $J$ und $M$ für unser $J_S$:
\begin{enumerate}
\item $Y$ bis zur Tiefe $J$ zerlegen,
\item $a_J$ durch den Nullvektor ersetzen,
\item die Details $d_1, \dots, d_{J_S}$ schwellen, $J_S < J$,
\item aus dem Nullvektor, den unveränderten $d_{J_S+1}, \dots, d_J$ und den
  geschwellten $d_1, \dots, d_{J_S}$ rekonstruieren.
\end{enumerate}
Als Formel:

\begin{Definition}[Ein-Schritt-Verfahren]
\label{def:ein-schritt}
Zu Tiefen $1 \le J_S < J$ und einer Schwelle $T > 0$ sei
\begin{equation}
\label{eq:ein-schritt}
  \hat f = A^\top\bigl(0,\; d_J, \dots, d_{J_S+1},\;
  \rho^{\mathrm S}_T(d_{J_S}), \dots, \rho^{\mathrm S}_T(d_1)\bigr),
\end{equation}
wobei das Soft-Thresholding $\rho^{\mathrm S}_T$ aus
Definition~\ref{def:thresholding} komponentenweise angewandt wird.
\end{Definition}

Da $A$ orthogonal ist, ist $A^\top = A^{-1}$ die Rekonstruktion
(Abschnitt~\ref{sec:onb-invarianz}); ohne Änderung der Koeffizienten wäre
$A^\top(AY) = Y$.

Soft- statt Hard-Thresholding verwenden wir wie Castillo et al., weil
Hard-Thresholding Unstetigkeiten erzeugt, Soft-Thresholding nicht
\cite[Abschn.~3.1.6]{castillo}; % S. 8: "The hard procedure creates discontinuities, while the soft procedure does not."
auch Alessio schwellt weich \cite[Kap.~15.2.1]{alessio}. % S. 719
Hard-Thresholding rechnen wir in Abschnitt~\ref{sec:test} zum Vergleich
mit. Gegenüber Algorithmus~\ref{alg:visushrink} ändern sich zwei Dinge: $a_J$
wird nicht geschwellt, sondern gelöscht, und die gröberen Details bleiben
unverändert.
 
Zu wählen sind noch die Tiefen $J$ und $J_S$ und die Schwelle $T$.

\subsection{Tiefe $J$ für die Drift}
\label{sec:tiefe-J}

Da $a_J$ gelöscht wird, soll es die Drift enthalten, aber keinen Anteil des
EKG. Nach \eqref{eq:frequenzbaender} erfasst $a_J$ die Frequenzen
$[0, F_{\max}/2^J]$. Die tiefste Frequenz des EKG ist die langsamste
Herzfrequenz \cite[Kap.~7.1]{sornmo}; % S. 458--459
die Grenze muss also zwischen Drift und Herzfrequenz liegen:
\begin{equation}
\label{eq:j-bedingung}
  f_{\mathrm{Drift}} \;<\; \frac{F_{\max}}{2^J} \;<\; f_{\mathrm{Herz}} .
\end{equation}
Castillo et al.\ und Alessio wählen dazu eine Grenze $f_c$, setzen
$F_{\max}/2^J \approx f_c$ und erhalten
\begin{equation}
\label{eq:lbw}
  J_{BW} = \operatorname{int}\Bigl[\log_2\Bigl(\frac{F_{\max}}{f_c}\Bigr)\Bigr],
\end{equation}
wobei $\operatorname{int}$ auf die nächste ganze Zahl rundet
\cite[Abschn.~3.1.2, Gl.~(9)]{castillo}, % S. 5; dort L_BW
\cite[Kap.~15.2.4]{alessio}. % S. 730
Bei Castillo et al.\ ist $F_{\max} = 125$\,Hz und $f_c = 1$\,Hz, also
$J_{BW} = 7$, und $a_7$ erfasst $0$ bis $0{,}977$\,Hz. % Castillo S. 5

\paragraph{Unsere Daten.}
Die Basisliniendrift entsteht meist durch die Atmung und liegt dann bei
$0{,}15$ bis $0{,}3$\,Hz \cite[Abschn.~2.2.1]{castillo}. % S. 3: "BW usually comes from respiration at frequencies varying between 0.15 and 0.3 Hz"
Nach den Annotationen liegt die Herzfrequenz im Mittel bei $75$ bzw.\ $70$
Schlägen pro Minute und für keinen Schlag unter $53$ pro Minute, also
$f_{\mathrm{Herz}} > 53/60\,\mathrm{Hz} \approx 0{,}88$\,Hz
(Code in Anhang~\ref{anh:code-kap6}). Mit
$F_{\max} = 180$\,Hz gilt:
\begin{itemize}
\item $f_c = 1$\,Hz wie bei Castillo et al.\ ergibt
  $J = \operatorname{int}[\log_2 180] = \operatorname{int}[7{,}49] = 7$ und
  die Grenze $180/2^7 = 1{,}41$\,Hz $> 0{,}88$\,Hz. Die rechte Ungleichung in
  \eqref{eq:j-bedingung} ist verletzt; $a_7$ enthielte Anteile des EKG.
\item Sörnmo und Laguna empfehlen eine Grenze etwas unter der langsamsten
  Herzfrequenz, etwa $f_c = 0{,}5$\,Hz, und warnen, dass bei zu hoher
  Grenze eine mit der Herzfrequenz schwingende Störung entsteht
  \cite[Kap.~7.1]{sornmo}. % S. 458--459
  Das ergibt
     \[
    J = \operatorname{int}[\log_2 360] = 8 \quad (\log_2 360 \approx 8{,}49),
    \qquad
    0{,}3\,\mathrm{Hz} \;<\; \frac{180}{2^8}\,\mathrm{Hz} \approx 0{,}70\,\mathrm{Hz}
    \;<\; 0{,}88\,\mathrm{Hz},
    \]
  \eqref{eq:j-bedingung} ist also erfüllt.
\end{itemize}
Es kommt dabei nicht genau auf $0{,}5$\,Hz an: Für jedes
$f_c \in [0{,}50;\,0{,}99]$\,Hz ist $\log_2(180/f_c) \in [7{,}5;\,8{,}5]$,
also $J = 8$. Die Grenze darf aber nicht zu knapp über der Drift liegen:
Bei Castillo et al.\ ist für alle drei Abtastraten die Stufe mit der Grenze
$F_{\max}/2^L \approx 0{,}98$\,Hz am besten \cite[Abschn.~4, Tab.~1]{castillo}; % S. 8 (Tab. 1), S. 10: "adequate decomposition level ... 7, 8, and 9 ... corroborated by Table 1"
die nächsttiefere mit $0{,}49$\,Hz entfernt Drift bis $0{,}31$\,Hz meist
deutlich schlechter. Die Grenzfrequenzen sind aus $F_{\max}/2^L$ berechnet.
Unsere Grenze $0{,}70$\,Hz liegt mehr als doppelt so hoch wie die Drift. In
Abschnitt~\ref{sec:test} vergleichen wir $J = 7, 8, 9$ am Test-EKG.

\subsection{Anzahl $J_S$ der geschwellten Stufen}

$J_S$ legt fest, wie viele der feinsten Stufen geschwellt werden: $d_1$ bis
$d_{J_S}$ werden geschwellt, $d_{J_S+1}$ bis $d_J$ bleiben unverändert.
Nach Lemma~\ref{lem:rauschen-invariant} hat jeder Koeffizient von
$\sigma AZ$ den Erwartungswert $0$ und die Varianz $\sigma^2$. 
Die erwartete Energie des Rauschens ist damit
\[
  \mathbb E\,\norm{\sigma AZ}^2
  = \sum_{m=0}^{N-1} \mathbb E\bigl[(\sigma AZ)_m^2\bigr]
  = \sigma^2 N ,
\]
und auf $d_j$ mit
\anmerkung{normen konventinnen überarbeiten hier euklidisch aber in 3 ist ja L2..}
$N/2^j$ Koeffizienten entfällt der Teil $\sigma^2 N/2^j$. In den unveränderten Stufen 
$d_{J_S+1}$ bis $d_J$ bleibt damit im Mittel der Anteil
\begin{align*}
  \frac{1}{\sigma^2 N}\sum_{j=J_S+1}^{J} \sigma^2\,\frac{N}{2^j}
  &= \sum_{j=J_S+1}^{J} 2^{-j}
   = 2^{-J_S}\sum_{k=1}^{J-J_S} 2^{-k} \\
  &= 2^{-J_S}\bigl(1 - 2^{-(J-J_S)}\bigr)
   = 2^{-J_S} - 2^{-J}
\end{align*}
der Rauschenergie erhalten. Dabei wurde $j = J_S + k$ gesetzt und die
geometrische Summe $\sum_{k=1}^{n} 2^{-k} = 1 - 2^{-n}$ verwendet.
Ein grö\ss eres $J_S$ entfernt also mehr Rauschen,
schwellt aber auch Stufen, in denen das EKG liegt.

Castillo et al.\ schwellen bei $f_s = 500$\,Hz die Stufen $d_1$ bis $d_3$,
weil dort der Rauschanteil gro\ss{} ist und die gröberen Stufen die
wesentlichen Teile des EKG enthalten \cite[Abschn.~3.1.3]{castillo}. % S. 6: "noise content is significant in detail subbands d1, d2, and d3 ... only d1, d2, and d3 should be treated"
Wegen $F_{\max} = 250$\,Hz wird dort nach \eqref{eq:frequenzbaender} alles
oberhalb von
\[
  f_S := \frac{250\,\mathrm{Hz}}{2^3} = 31{,}25\,\mathrm{Hz}
\]
geschwellt.
Da dieselbe Zahl von Stufen bei anderer Abtastrate andere Frequenzen
bedeutet, übertragen wir nicht die Zahl der Stufen, sondern diese Grenze
(eigene Festlegung): Wir wählen das grö\ss te $J_S$, dessen geschwellte
Stufen oberhalb von $f_S$ liegen,
\[
  \frac{F_{\max}}{2^{J_S}} \ge f_S
  \iff J_S \le \log_2\frac{180}{31{,}25} \approx 2{,}53,
  \qquad\text{also } J_S = 2 .
\]
Die geschwellten Stufen $d_1$ und $d_2$ beginnen damit bei $45$\,Hz und
liegen insbesondere oberhalb von P- und T-Welle, die bis $20$\,Hz reichen
\cite[Kap.~2.3.1]{rebstadt}. % S. 17: "zwischen einem und 20 Hertz [62]"
Es bleiben $2^{-2} - 2^{-8} = 63/256 \approx 24{,}6\,\%$ der
Rauschenergie erhalten.
In $d_1$ und $d_2$ liegen neben Netzbrummen und Muskelrauschen auch die
hochfrequenten Anteile des QRS-Komplexes: Das EKG-Spektrum reicht bis
$150$\,Hz \cite[Kap.~2.4.3.2]{gacek}, % S. 33
und nur P- und T-Welle und die tieffrequenten Anteile des QRS-Komplexes
liegen unter $20$\,Hz \cite[Kap.~2.3.1]{rebstadt}. % S. 17
Da das Rauschen gleichmä\ss ig auf alle Detailkoeffizienten verteilt ist, das
EKG dagegen durch wenige gro\ss e Koeffizienten dargestellt wird, entfernt
Schwellen das Rauschen und erhält die Spitzen des QRS-Komplexes
\cite[Abschn.~2.2.2]{castillo}; % S. 3--4
diese Stufen werden deshalb geschwellt statt gelöscht.

\subsection{Schwelle}

Als Schwelle verwenden wir die universelle Schwelle
\eqref{eq:universelle-schwelle} mit dem Schätzer \eqref{eq:mad},
$T = \hat\sigma\sqrt{2\ln N}$, wie Castillo et al.\ und Alessio
\cite[Abschn.~3.1.4, Gl.~(10); 3.1.5, Gl.~(19), (23)]{castillo}, % S. 6--7; dort 1,4826 = 1/0,6745
\cite[Kap.~15.2.2, 15.2.3]{alessio}. % S. 721--722 
Geschätzt wird $\hat\sigma$ aus $d_1$: Bei wei\ss em Rauschen kann man die Standardabweichung aus den Koeffizienten
der ersten Stufe schätzen, die bis auf wenige gro\ss e Werte nur Rauschen
enthalten, und für alle Stufen verwenden 
\cite[Abschn.~3.1.5]{castillo}, % S. 7
\cite[Kap.~15.2.2]{alessio}. % S. 720--721
Für $N$ zwischen $2^6$ und $2^{19}$ überschreitet reines Rauschen diese
Schwelle nur in etwa einem Zehntel der Realisierungen
\cite[Kap.~15.2.3]{alessio}. % S. 723
Gerechnet wurde mit PyWavelets \cite{pywavelets} (Code in 
Anhang~\ref{anh:code-kap6}).

\paragraph{Zusammenfassung.}
Mit den gewählten Parametern lautet das Verfahren:

\begin{Definition}[Verfahren zur Entrauschung von EKG-Signalen]
\label{def:verfahren}
Gegeben seien Messwerte $Y\in\R^N$ mit Abtastrate $f_s = 360$\,Hz, wobei
$N$ durch $2^8$ teilbar ist. Das entrauschte Signal $\hat f\in\R^N$ entsteht
in vier Schritten:
\begin{enumerate}
\item \emph{Zerlegen:} $AY = (a_8, d_8, \dots, d_1)$ mit der periodisierten
  DWT der Tiefe $J = 8$ und dem Wavelet $\psi_6$ (Abschnitt~\ref{sec:tiefe-J}).
\item \emph{Schwelle bestimmen:}
  $\hat\sigma$ nach \eqref{eq:mad} aus $d_1$ und
  $T = \hat\sigma\sqrt{2\ln N}$.
\item \emph{Koeffizienten ändern:} $a_8$ durch $0$ ersetzen, $d_1$ und $d_2$
  mit $\rho^{\mathrm S}_T$ schwellen ($J_S = 2$), $d_3$ bis $d_8$ unverändert
  lassen.
\item \emph{Rekonstruieren:}
  \[
    \hat f = A^\top\bigl(0,\; d_8, \dots, d_3,\;
    \rho^{\mathrm S}_T(d_2),\; \rho^{\mathrm S}_T(d_1)\bigr).
  \]
\end{enumerate}
Das ist das Ein-Schritt-Verfahren aus Definition~\ref{def:ein-schritt} mit
$J = 8$, $J_S = 2$ und der universellen Schwelle.
\end{Definition}

Im Folgenden testen wir das Verfahren zuerst an einem künstlichen EKG, bei
dem $f$ bekannt ist (Abschnitt~\ref{sec:test}), und wenden es dann auf die
Datensätze 100 und 103 an (Abschnitt~\ref{sec:ergebnisse}).

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Test an einem künstlichen EKG}
\label{sec:test}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Bei echten Aufzeichnungen ist $f$ unbekannt, und ein quantitatives Gütema\ss{}
lässt sich nur schwer angeben; Castillo et al.\ bewerten ihr Verfahren deshalb
an künstlichen EKG \cite[Abschn.~4]{castillo}. % S. 10
Wir gehen ebenso vor.

Als Grundlage dient ein EKG $f_{\mathrm{syn}}\in\R^N$ aus dem dynamischen
Modell von McSharry et al.\ \cite{mcsharry}, berechnet mit der Funktion
\texttt{ecg\_simulate} (Methode \texttt{ecgsyn}) aus NeuroKit2
\cite{neurokit2} für $f_s = 360$\,Hz, eine Herzfrequenz von $70$ Schlägen
pro Minute und $N = 2^{16}$ Abtastwerte, also etwa $3$\,min. Das Modell
enthält von sich aus eine Atmungsdrift \cite[Abschn.~IV, Gl.~(2)]{mcsharry}; % S. 291
diese haben wir abgeschaltet, damit das Test-EKG selbst keine Drift enthält.

Für $x\in\R^N$ sei $\bar x = \frac1N\sum_{n=0}^{N-1} x_n$ der Mittelwert und
$\mathbf 1 = (1,\dots,1)\in\R^N$, also $\scal{x}{\mathbf 1} = N\bar x$.

\begin{Lemma}
\label{lem:mittelwert-null}
Jedes $\hat f$ nach \eqref{eq:ein-schritt} hat den Mittelwert $\bar{\hat f} = 0$.
\end{Lemma}

\begin{proof}
Setzt man $\omega = 0$ in \eqref{eq:symbol-H} ein, so ist
$H(0) = 2^{-1/2}\sum_k h_k$, wegen $H(0) = 1$ nach \eqref{eq:qmf} also
$\sum_k h_k = \sqrt2$. Ebenso ist $G(0) = 2^{-1/2}\sum_k g_k$, und nach
\eqref{eq:qmf} und \eqref{eq:G-aus-H} gilt $G(0) = 0$
(Abschnitt~\ref{sec:fourier-bedingungen}, Aussage~(b)), also
$\sum_k g_k = 0$. Für einen konstanten Vektor $a_{j-1,n} = c$ liefert
\eqref{eq:analyse} daher
\[
  a_{j,k} = c\sum_n h_{n-2k} = \sqrt2\,c, \qquad
  d_{j,k} = c\sum_n g_{n-2k} = 0 ,
\]
bei periodischer Fortsetzung durchläuft $n-2k$ dabei jeden Index des Filters
genau einmal. Eine Stufe bildet einen konstanten Vektor also auf einen
konstanten Vektor in der Approximation und auf den Nullvektor in den Details
ab. Wiederholt man das $J$-mal, ist $A\mathbf 1$ nur im Teil $a_J$ von null
verschieden. In \eqref{eq:ein-schritt} ist $\hat f = A^\top u$, wobei $u$ im
Teil $a_J$ null ist. Damit ist
\[
  N\,\bar{\hat f} = \scal{\hat f}{\mathbf 1} = \scal{A^\top u}{\mathbf 1}
  = \scal{u}{A\mathbf 1} = 0 . \qedhere
\]
\end{proof}

Einen konstanten Versatz kann das Verfahren also nicht wiedergeben; er ist
für das EKG ohne Bedeutung, da dessen Grö\ss en gegenüber der isoelektrischen
Linie gemessen werden (Abschnitt~\ref{sec:ekg}). Als Test-EKG wählen wir
deshalb das mittelwertfreie
\[
  f := f_{\mathrm{syn}} - \bar f_{\mathrm{syn}}\,\mathbf 1
\]
(eigene Festlegung). Wie Castillo et al.\ verwenden wir eine sinusförmige
Drift und Gauß'sches weißes Rauschen, addieren beide aber gemeinsam
(vgl.\ \cite[Abschn.~4]{castillo}): % S. 10--11; dort Drift und Rauschen getrennt
\[
  Y = f + b + \sigma Z, \qquad b(t) = 0{,}5\,\mathrm{mV}\cdot\sin(2\pi\cdot 0{,}25\,\mathrm{Hz}\cdot t),
\]
mit $\sigma = 0{,}025$\,mV; mit diesem Wert bilden McSharry et al.\ das
Messrauschen eines echten EKG nach \cite[Abschn.~V]{mcsharry}. % S. 292: "normally distributed measurement errors with mean zero and standard deviation 0.025 mV"
Dabei ist $Z\in\R^N$ eine feste
Realisierung, also ein Vektor mit $N$ standardnormalverteilten Einträgen.
Abbildung~\ref{fig:testsignal} zeigt einen Ausschnitt von $4$\,s: oben das

\section{Diskussion}
\label{sec:diskussion}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Am Test-EKG sinkt die Fehlerenergie von $8236{,}4$ auf $18{,}1$, und $J = 8$ ist deutlich besser als $J = 7$ und $J = 9$, wie es
\eqref{eq:j-bedingung} erwarten lässt (Tabelle~\ref{tab:test}). An den
Datensätzen $100$ und $103$ ist die Drift sichtbar entfernt, die Form der
Schläge bleibt erhalten (Abbildung~\ref{fig:vergleich}). Dazu braucht es beide
Bausteine: Das Thresholding aus Kapitel~\ref{chapter:denoising} erreicht die
Drift nicht, denn ihre Energie steckt in den Approximationskoeffizienten
$a_J$, und die lässt das Thresholding unverändert; erst die Wahl von $J$ nach
den Frequenzen von Drift und Herzschlag erlaubt, $a_J$ gezielt zu entfernen.
Beides geschieht in einer einzigen Zerlegung (Abschnitt~\ref{sec:verfahren}).

Die Ergebnisse haben folgende Grenzen.
\begin{itemize}
\item Die Aussage über den Fehler gilt nur für das Test-EKG. Es enthält nur
  eine Sinusdrift und wei\ss es Rauschen, kein Netzbrummen und kein
  Muskelrauschen. Die echten Aufzeichnungen beurteilen wir nur nach Augenschein.
  Ob die Änderungen am EKG klinisch unbedenklich sind, müsste ärztlich
  beurteilt werden; ohne klinische Erfahrung kann ein Verfahren unzulässige
  Verzerrungen einführen \cite[Kap.~1.4.1]{sornmo}. % S. 20
\item Das Nullsetzen von $a_8$ entfernt alles unter $0{,}70$\,Hz, auch
  langsame Anteile des EKG: Dessen Spektrum reicht nach unten bis
  $0{,}005$\,Hz \cite[Kap.~2.4.3.2]{gacek}, % S. 33
  bei den MIT-BIH-Daten, die auf $0{,}1$ bis $100$\,Hz bandbegrenzt sind,
  bis $0{,}1$\,Hz (Abschnitt~\ref{sec:daten}).
  Bei langsamem Herzschlag unter $42$ Schlägen pro Minute läge die
  Herzfrequenz selbst unter $0{,}70$\,Hz; Sörnmo und Laguna nennen für
  Bradykardie etwa $40$ Schläge pro Minute \cite[Kap.~7.1]{sornmo}. % S. 458
  Dann wäre $J = 9$ zu wählen, mit mehr verbleibender Drift.
\item Elektrodenbewegung liegt bei $1$ bis $10$\,Hz und damit in den
  unveränderten Stufen; das Verfahren kann sie nicht entfernen. Ob der Abfall
  bei $1090$\,s in Datensatz~$103$ von einer Elektrodenbewegung stammt, lässt
  sich aus den Daten nicht sagen; das Directory markiert den Abschnitt nur als
  verrauscht. Er bleibt als Einbruch sichtbar.
\item Die Schwelle beruht auf einem einzigen $\hat\sigma$ aus $d_1$ und setzt
  wei\ss es Rauschen voraus. Für Rauschen, das nicht wei\ss{} ist, schätzt man
  $\sigma$ auf jeder Stufe einzeln \cite[Kap.~15.2.2]{alessio}, % S. 721
  \cite[Abschn.~3.1.5]{castillo}. % S. 7
  Muskelrauschen liegt bei $20$ bis $80$\,Hz (Tabelle~\ref{tab:stoerungen}),
  also vor allem in $d_2$ und $d_3$. Auf $d_3$ bleibt es unverändert stehen,
  und auf $d_2$ ist die aus $d_1$ geschätzte Schwelle womöglich zu klein;
  das Muskelrauschen wird also nur unvollständig entfernt.
\item $J_S = 2$ beruht auf der von Castillo et al.\ übertragenen Grenze
  $f_S = 31{,}25$\,Hz (Abschnitt~\ref{sec:verfahren}) und bestätigt sich am
  Test-EKG. Welches $J_S$ am besten ist, hängt aber vom Verhältnis von EKG
  und Rauschen auf $d_3$ ab: Bei $\sigma = 0{,}01$ und $0{,}05$\,mV ist
  ebenfalls $J_S = 2$ besser ($9{,}7$ statt $11{,}7$ und $48{,}2$ statt
  $52{,}0$), bei $\sigma = 0{,}1$\,mV dagegen $J_S = 3$ ($116{,}1$ statt
  $168{,}1$; eigene Rechnung, Listing~\ref{lst:kap6-test}).
  Bei den Datensätzen $100$ und $103$ ist $\hat\sigma \approx 0{,}006$\,mV
  (Listing~\ref{lst:kap6-vergleich}), das wei\ss e Rauschen also schwächer
  als im Test; für dieses spricht das ebenfalls für $J_S = 2$.
\item Die Datenbank wurde für die Bewertung von Arrhythmie-Detektoren angelegt
  \cite{mitbih}, % S. 45 "test material for evaluation of arrhythmia detectors"
  nicht für das Entrauschen; eine Datenbank für Fragen zu nutzen, für die sie
  nicht gedacht war, ist ein bekanntes Risiko \cite[Kap.~1.4.1]{sornmo}. % S. 20--21
  Zudem verwenden wir nur zwei Aufzeichnungen. Die Parameter sind allerdings
  aus der Literatur übernommen und nicht an diesen Daten angepasst.
\end{itemize}

Simulierte Signale, bei denen das wahre Signal bekannt ist, erlauben eine
Genauigkeitsaussage wie in Abschnitt~\ref{sec:test}; den Hauptteil der
Bewertung müssen aber gemessene Aufzeichnungen tragen
\cite[Kap.~1.4.2]{sornmo}. % S. 21--22

\section{Zulässigkeitskonstante des Haar-Wavelets}
\label{anh:haar-konstante}
%-------------------------------------------------------------------------------
 
Wir berechnen $C_{\psi^{\mathrm H}}$ aus \eqref{eq:haar-fourier}. Für
$\omega \neq 0$ ist $\sin(\omega/4)\operatorname{sinc}(\omega/4) = 4\sin^2(\omega/4)/\omega$,
also
\[
  \bigl\lvert\widehat{\psi^{\mathrm{H}}}(\omega)\bigr\rvert^{2}
  = \frac{16\sin^{4}(\omega/4)}{2\pi\,\omega^{2}} .
\]
Dieser Ausdruck ist gerade in $\omega$. Mit der Substitution $u = \omega/4$
folgt
\[
  C_{\psi^{\mathrm H}}
  = 2\pi \cdot 2 \int_0^\infty \frac{16\sin^{4}(\omega/4)}{2\pi\,\omega^{3}}\,d\omega
  = 32\int_0^\infty \frac{\sin^{4}(\omega/4)}{\omega^{3}}\,d\omega
  = 2\int_0^\infty \frac{\sin^{4}u}{u^{3}}\,du .
\]
Zur Auswertung schreiben wir $8\sin^4 u = g(u)$ mit
$g(u) := 3 - 4\cos(2u) + \cos(4u)$. Die Taylorentwicklung liefert
$g(u) = 8u^4 + O(u^6)$ für $u \to 0$. Zweimalige partielle Integration ist
daher ohne Randterme möglich und ergibt
\[
  \int_0^\infty \frac{g(u)}{u^{3}}\,du
  = \frac12\int_0^\infty \frac{g''(u)}{u}\,du
  = 8\int_0^\infty \frac{\cos(2u) - \cos(4u)}{u}\,du
  = 8\ln 2 .
\]
Im letzten Schritt wurde das Frullani-Integral
$\int_0^\infty \bigl(\cos(\alpha u) - \cos(\beta u)\bigr)/u\,du = \ln(\beta/\alpha)$
für $\alpha,\beta > 0$ verwendet. Also ist
$\int_0^\infty \sin^4(u)/u^3\,du = \ln 2$ und
$C_{\psi^{\mathrm H}} = 2\ln 2$, in Übereinstimmung mit
\cite[Kap.~1.1, Beispiel~(1)]{lmr}. % S. 18--19

\section{Fourier-Transformierte des Testsignals}
\label{anh:fourier-testsignal}
%-------------------------------------------------------------------------------
 
Wir berechnen $\hat f$ für das Testsignal aus Beispiel~\ref{bsp:testsignal}.
Wegen $f\in L^1(\R)$ ist $\hat f$ punktweise durch das Integral gegeben. In
der Konvention aus Bemerkung~\ref{bem:fourier-konvention} ist
\[
  \hat f(\omega)
  =\frac{1}{\sqrt{2\pi}}\left(
      \int_0^3\bigl(1-\cos(2\pi t)\bigr)e^{-i\omega t}\,dt
      +\frac12\int_4^5\bigl(1-\cos(10\pi t)\bigr)e^{-i\omega t}\,dt
    \right).
\]
Sei zunächst $\omega\notin\{0,\pm2\pi,\pm10\pi\}$. Benutzt werden nur
$\cos x=\tfrac12\bigl(e^{ix}+e^{-ix}\bigr)$ und
\begin{equation}\label{eq:stammfunktion-exp}
  \int_\alpha^\beta e^{ct}\,dt=\frac{1}{c}\bigl(e^{c\beta}-e^{c\alpha}\bigr),
  \qquad c\in\C\setminus\{0\}.
\end{equation}
 
\paragraph{Die langsame Komponente.}
Mit $q_1:=1-e^{-3i\omega}$ zerlegen wir
\[
  \int_0^3\bigl(1-\cos(2\pi t)\bigr)e^{-i\omega t}\,dt
  =\underbrace{\int_0^3 e^{-i\omega t}\,dt}_{=:A_1}
   -\underbrace{\int_0^3 \cos(2\pi t)\,e^{-i\omega t}\,dt}_{=:B_1}.
\]
Nach \eqref{eq:stammfunktion-exp} mit $c=-i\omega$ ist $A_1 = q_1/(i\omega)$.
Für $B_1$ schreiben wir
$\cos(2\pi t)e^{-i\omega t}=\tfrac12\bigl(e^{i(2\pi-\omega)t}+e^{-i(2\pi+\omega)t}\bigr)$.
Nach \eqref{eq:stammfunktion-exp} ist
\[
  B_1=\frac{e^{3i(2\pi-\omega)}-1}{2i(2\pi-\omega)}
     +\frac{e^{-3i(2\pi+\omega)}-1}{-2i(2\pi+\omega)}.
\]
Auf $[0,3]$ liegen genau drei volle Perioden. Daher ist $e^{\pm 6\pi i}=1$ und
$e^{3i(2\pi-\omega)}=e^{-3i(2\pi+\omega)}=e^{-3i\omega}$. Beide Zähler werden
zu $-q_1$, und es folgt
\[
  B_1=\frac{q_1}{2i}\left(\frac{1}{2\pi+\omega}-\frac{1}{2\pi-\omega}\right)
     =\frac{q_1\,\omega}{i\,(\omega^2-4\pi^2)} .
\]
Mit $1/i=-i$ ergibt sich
\[
  A_1-B_1
  =\frac{q_1}{i}\left(\frac1\omega-\frac{\omega}{\omega^2-4\pi^2}\right)
  =\frac{i\,4\pi^2\,q_1}{\omega\,(\omega^2-4\pi^2)} .
\]
 
\paragraph{Die schnelle Komponente.}
Mit $q_2:=e^{-4i\omega}-e^{-5i\omega}$ verläuft die Rechnung gleich, mit
$10\pi$ statt $2\pi$ und den Grenzen $4$ und $5$. An die Stelle von
$e^{\pm6\pi i}=1$ treten $e^{\pm40\pi i}=e^{\pm50\pi i}=1$. Mit dem
Amplitudenfaktor $\tfrac12$ erhält man
\[
  \frac12\left(\int_4^5 e^{-i\omega t}\,dt - \int_4^5 \cos(10\pi t)\,e^{-i\omega t}\,dt\right)
  =\frac{i\,50\pi^2\,q_2}{\omega\,(\omega^2-100\pi^2)} .
\]
 
\paragraph{Ergebnis.}
Addition beider Anteile und der Vorfaktor $(2\pi)^{-1/2}$ ergeben
\begin{equation}\label{eq:ft-testsignal}
  \hat f(\omega)
  =\frac{i}{\sqrt{2\pi}\,\omega}
   \left[
     \frac{4\pi^{2}\bigl(1-e^{-3i\omega}\bigr)}{\omega^{2}-4\pi^{2}}
     +\frac{50\pi^{2}\bigl(e^{-4i\omega}-e^{-5i\omega}\bigr)}{\omega^{2}-100\pi^{2}}
   \right].
\end{equation}
Da $\hat f$ als Fourier-Transformierte einer $L^1$-Funktion stetig ist
\cite[Satz~12.1]{serov}, gilt \eqref{eq:ft-testsignal} an den Stellen
$\omega\in\{0,\pm2\pi,\pm10\pi\}$ im Sinne des Grenzwerts. Direkt aus dem
Integral folgen
\[
  |\hat f(0)|=\frac{7}{2\sqrt{2\pi}},\qquad
  |\hat f(\pm2\pi)|=\frac{3}{2\sqrt{2\pi}},\qquad
  |\hat f(\pm10\pi)|=\frac{1}{4\sqrt{2\pi}} .
\]
Zur Kontrolle wurde \eqref{eq:ft-testsignal} an mehreren Stellen mit
numerischer Quadratur des Integrals verglichen
(Anhang~\ref{anh:code-testsignal}).

\section{Stammfunktion des Testsignals}
\label{anh:stammfunktion-testsignal}
%-------------------------------------------------------------------------------
 
Für die Stammfunktion $F(t):=\int_{-\infty}^{t}f(s)\,ds$ des Testsignals gilt
\[
  F(t)=
  \begin{cases}
    0, & t<0,\\[2pt]
    t-\dfrac{\sin(2\pi t)}{2\pi}, & 0\le t\le 3,\\[4pt]
    3, & 3\le t\le 4,\\[2pt]
    3+\dfrac{t-4}{2}-\dfrac{\sin(10\pi t)}{20\pi}, & 4\le t\le 5,\\[4pt]
    \dfrac72, & t>5.
  \end{cases}
\]
Auf $[0,3]$ folgt das durch direkte Integration. Wegen $\sin(6\pi)=0$ ist
$F(3)=3$. Auf $[3,4]$ verschwindet $f$. Auf $[4,5]$ ist
\[
  F(t)=3+\frac12\int_4^t\bigl(1-\cos(10\pi s)\bigr)ds
      =3+\frac{t-4}{2}-\frac{\sin(10\pi t)}{20\pi},
\]
da $\sin(40\pi)=0$ ist. Wegen $\sin(50\pi)=0$ ist $F(5)=\tfrac72=\int_\R f$.
Das passt zu $\hat f(0)=(2\pi)^{-1/2}\cdot\tfrac72$.
